\begin{tikzpicture}[scale=1.3]
    % 4x4 grid graph
    \begin{scope}[shift={(-3,0)}]
        \foreach \i in {0,1,2,3} {
            \foreach \j in {0,1,2, 3} {
                \node[v_point] (G\i\j) at (\i,\j) {};
            }
        }
        % horizontal edges
        \foreach \i in {0,1, 2} {
            \foreach \j in {0,1,2, 3} {
                \draw[new_edge] (G\i\j) -- (G\the\numexpr\i+1\relax\j);
            }
        }
        % vertical edges
        \foreach \i in {0,1,2,3} {
            \foreach \j in {0,1, 2} {
                \draw[new_edge] (G\i\j) -- (G\i\the\numexpr\j+1\relax);
            }
        }
    \end{scope}
    % Tree graph
    \begin{scope}[shift={(2,0)}]
        % vertices
        \node[v_point] (T0) at (2,3) {};
        \node[v_point] (T1) at (1,2) {};
        \node[v_point] (T2) at (3,2) {};
        \node[v_point] (T3) at (0,1) {};
        \node[v_point] (T4) at (2,1) {};
        \node[v_point] (T5) at (4,1) {};
        \node[v_point] (T6) at (2,0) {};
        % edges
        \draw[new_edge] (T0) -- (T1);
        \draw[new_edge] (T0) -- (T2);
        \draw[new_edge] (T1) -- (T3);
        \draw[new_edge] (T1) -- (T4);
        \draw[new_edge] (T2) -- (T5);
        \draw[new_edge] (T4) -- (T6);
    \end{scope}
\end{tikzpicture}
