\begin{trditev}\label{trd:avtomorfizem_ohranja_razdalje}
    Naj bo $G$ graf in $\aut(G)$ grupa njegovih avtomorfizmov. Potem 
    vsak avtomorfizem $g \in \aut(G)$ ohranja razdalje, torej $d(g\cdot u, g\cdot v) = d(u, v)$ 
    za vsak par vozlišč $u$ in $v$ v $G$.
\end{trditev}
\begin{dokaz}
    Naj bosta $u$ in $v$ dve vozlišči v grafu $G$ na razdalji $d(u, v) = k$ in 
    $P = u u_1 \cdots u_{k-1} v$ najkrajša pot med njima.
    Potem je $g\cdot P = g\cdot u \; g\cdot u_1 \cdots g\cdot u_{k-1} \; g\cdot v$ pot med $g\cdot u$ in $g\cdot v$,
    in ker je $P$ najkrajša pot med $u$ in $v$, je tudi $g\cdot P$ najkrajša pot med $g\cdot u$ in $g\cdot v$,
    torej $d(g\cdot u, g\cdot v) = k$.
\end{dokaz}

\begin{definicija}[Binarna relacija]\label{def:binarna_relacija} % NUJNO
    Binarna relacija $R$ na množicah $X$ in $Y$ je podmnožica kartezičnega produkta 
    $X \times Y = \{(x, y)\mid x\in X \land y\in Y\}$. $(x, y) \in R$ razumemo kot
    $x$ je v relaciji z $y$, in pišemo $xRy$.
\end{definicija}

\begin{definicija}[Ekvivalenčna relacija]\label{def:ekvivalencna_relacija} % NUJNO
Ekvivalenčna relacija $R \subseteq X\times X$ je binarna relacija (\ref{def:binarna_relacija}), ki je refleksivna,
simetrična in tranzitivna. Drugače povedano, je to binarna relacija, ki zadošča naslednjim pogojem.
\begin{enumerate}[label=(\roman*)]
    \item $\forall x \in X:\; xRx$. (refleksivnost)
    \item $\forall x, y \in X:\; xRy \implies yRx$. (simetričnost)
    \item $\forall x, y, z \in X:\; (xRy \land yRz) \implies xRz$. (tranzitivnost)
\end{enumerate}
\end{definicija}

\begin{definicija}[Ekvivalenčni razredi]\label{def:ekvivalencni_razredi} % NUJNO
    Naj bo $R$ ekvivalenčna relacija na množici $X$. Ekvivalenčni razred elementa $x \in X$ je množica 
    $[x] = \{y \in X \mid xRy\}$ vseh elementov, ki so v relaciji z $x$.

    Set vseh ekvivalenčnih razredov $X / R = \{[x]\mid x \in X\}$ tvori particijo za $X$.
\end{definicija}

\begin{trditev}
    Naj bo $\lambda$ lastna vrednost matrike $A$, potem obstaja tudi $\theta$ lastna
    vrednost matrike $A^*A$, da je $\lambda\lambda^* = \theta$.
    Seveda je to očitno, saj 
    $A^*A\mathbf{v} = A^*(A\mathbf{v}) = A^*(\lambda \mathbf{v})
     = \lambda A^*\mathbf{v} = \lambda \lambda^* \mathbf{v}$.
\end{trditev}

\begin{trditev}
    Velja $\sled(AB) = \sled(BA)$ za vsaka $A, B$, seveda ker ... (prek vsote ali baznih matrik).
    Sled matrike $A$ je enaka sledi njene podobne matrike $B$. 
    Očitno res, saj $\exists B: \; B = P^{-1}AP$, 
    in sledi $B$ je $\sled(B) = \sled(P^{-1}AP) = \sled(APP^{-1}) = \sled(A)$.
\end{trditev}

\begin{trditev}
    Naj bo $\theta$ enostavna ničla polinoma $f(a)$ z lastnostjo $f(M) = 0$, 
    naj bo dalje $m(\theta)$ večkratnost $\theta$ kot lastne vrednosti matrike $M$.
    Potem velja 
    \begin{equation*}
        m(\theta) = \frac{\sled(f_{\theta}(M))}{f'(\theta)},
    \end{equation*}
    kjer je $f_{\theta}(a) = \frac{f(a)}{a-\theta}$.
\end{trditev}
\begin{dokaz}
    Ker za vsako matriko obstaja Jordanova dekompozicija, očitno velja 
    \[\sled(f_{\theta}(M)) = \sled(f_{\theta}(P^{-1}JP)) = \sled(f_{\theta}(J)) = 
    \sum_{i=1}^n f_{\theta}(\lambda_i),\] 
    kjer so $\lambda_i$ lastne vrednosti matrike $M$. Ker pa je $f(M) = 0$, je tudi 
    $\sled(f(M)) = 0$ in zato za vse druge $\theta_i$ velja $f(\theta_i) = 0$, 
    torej je $\sled(f_{\theta}(M)) = m(\theta)f_{\theta}(\theta)$, kar je enako $m(\theta)f'(\theta)$, saj je $\theta$ enostavna ničla polinoma $f(a)$.
    Ker $\lim_{a \to \theta} f_{\theta}(a) = = f'(\theta)$, je zato
\end{dokaz}

\begin{definicija}
    Kompleksna kvadratna obrnljiva matrika $U$ je unitarna, če je njen inverz 
    enak njenemu konjugirano transponiranemu, torej $U^{-1} = U^*$. Oziroma,
    če je njen inverz enak njeni hermitiranki ($U^{-1} = U^H$).
\end{definicija}
\begin{opomba}
    Unitarna matrika $U$ ohranja skalarni produkt, 
    torej $\langle U\mathbf{x}, U\mathbf{y} \rangle = x^* U^*Uy = x^*Iy= \langle \mathbf{x}, \mathbf{y} \rangle$.
    Prav tako je $|\det(U)| = |\det(U)\det(U^*)|^{1/2} = |\det(I)|^{1/2} = 1$, 
    torej je $\det(U) = e^{i\theta}$ za nek $\theta \in \mathbb{R}$.
    Velja tudi to, da je vsaka unitarna matrika sestavljena zgolj iz rotacij in refleksij,
    saj je vsaka unitarna matrika diagonalizabilna z diagonalnimi elementi na enotski krožnici, %Cartan–Dieudonné theorem
\end{opomba}
\begin{definicija}
    Posebna unitarna grupa $SU(n)$ je podgrupa unitarne grupe $U(n)$, ki jo tvorijo 
    vse unitarne matrike z determinantno enako 1.
    \[SU(n) := \{U \in \mathbb{C}^{n\times n}\mid U^*U = I, \det(U) = 1\}\]
\end{definicija}
\begin{zgled}
    Ker zgolj skalarne matrike $\lambda I$ komutirajo z vsem ostalimi matrikami, je 
    center posebne unitarne grupe $SU(n)$ enak $Z(SU(n)) = \{\lambda I \mid \lambda^n = 1\}$.
    Posledično kvocient $SU(n) / Z(SU(n))$ povzroči razpad grupe $SU(n)$ na kosete
    $\{\lambda I A \mid \lambda I \in Z(G)\}$ ki tretirajo elemente kot enake če se razlikujejo
    za večkratnik. Ta kvocient imenujemo projektivna posebna unitarna grupa $PSU(n)$.
\end{zgled}
\begin{definicija}
    Kompleksna kvadratna matrika $U$ je anti-hermitska, če je njena konjugirana transponiranka
    enaka njeni negativni vrednosti, torej $U^* = -U$. 
\end{definicija}


\subsection{Osnovna teorija grafov}

\begin{definicija}
    Graf je razdaljno regularen, če je regularen, in če za vsak par vozlišč $u$ in $v$ velja, 
    da je število vozlišč razdalje $j$ od $u$ in $k$ od $v$ odvisno le od $j$, $k$ in razdalje med $u$ in $v$.

    Formalno je to $\exists s: \mathbb{N}_0^3 \to \mathbb{N}_0$ taka, da
    $\forall u, v \in V(G):\;s(j, k, d(u, v)) = |\{w \in V(G) \mid d(u, w) = j \land d(v, w) = k\}|$.
\end{definicija}

\begin{definicija}
    Presečno polje razdaljno regularnega grafa je zaporedje
    \[\{b_0, \ldots, b_{D-1}; c_1, \ldots, x_D\},\]
    kjer je $D$ premer grafa.
    Za vsak $1 \leq j \leq D$ je $b_j$ število sosedov vozlišča $u$, ki so na razdalji $j+1$
    od vozlišča $v$, in $c_j$ število sosedov od vozlišča $u$, ki so na razdalji $j-1$ od vozlišča $v$.

    Od tu lahko izračunamo tudi $a_j$, ki pove, koliko sosedov ima $u$, ki so na razdalji $j$ od vozlišča $v$,
    in sicer $a_j = k - b_j - c_j$, kjer je $k$ red grafa.
\end{definicija}
\begin{definicija}\label{def:presecna_matrika}
    Presečno polje lahko predstavimo tudi z presečno matriko $M$, in sicer tako, da izrazimo sosednostno matriko $A$
    v bazi vektorjev $\{\mathbf{e}_{N_0(v)}, \ldots, \mathbf{e}_{N_D(v)}\}$, kjer je $\mathbf{e}_{N_j(v)}$ 
    vektor, ki ima enice na pozicijah vozlišč, ki so na razdalji $j$ od vozlišča $v$, in ničle drugje.
    V tej bazi je, ob dogovoru $b_{-1} = c_{D+1} = 0$,
    $A\mathbf{e}_{N_j(v)} = b_{j-1}\mathbf{e}_{N_{j-1}(v)} + a_j \mathbf{e}_{N_j(v)} + c_{j+1} \mathbf{e}_{N_{j+1}(v)}$,
    in posledično z razvojem $A$ po taki bazi dobimo tridiagonalno matriko

    \begin{equation}
    M = \begin{bmatrix}
    a_0 & b_0 & 0 & \cdots & 0 \\
    c_1 & a_1 & b_1 & \cdots & 0 \\
    0 & c_2 & a_2 & \cdots & 0 \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & 0 & \cdots & b_{D-1} \\
    0 & 0 & c_D & \cdots & a_D
    \end{bmatrix}
    \end{equation}
\end{definicija}

\begin{opomba}
Ker je presečna matrika $M$ podobna matriki $A$ ($\exists P:\; M = P^{-1}AP$),
ima tudi enake lastne vrednosti. Velja celo to, da je minimalni polinom
matrike $M$ ravno minimalni polinom matrike $A$.
\end{opomba}

\subsubsection{Osnovne definicije}

\begin{definicija}[Grupa]
    Grupa je množica $G$ skupaj z binarno operacijo $\cdot$ na $G$, ki zadošča naslednjim pogojem:
    \begin{enumerate}[label=(\roman*)]
        \item $\forall a, b, c \in G:\; (a\cdot b)\cdot c = a\cdot (b\cdot c)$ (asociativnost)
        \item $\exists e \in G:\; \forall a \in G:\; e\cdot a = a\cdot e = a$ (obstoj enote)
        \item $\forall a \in G:\; \exists a^{-1} \in G:\; a\cdot a^{-1} = a^{-1}\cdot a = e$ (obstoj inverza)
    \end{enumerate}
\end{definicija}

\begin{definicija}[Podgrupa]\label{def:podgrupa}
    Podgrupa $H$ grupe $G$ je podmnožica $H \subseteq G$, ki je tudi sama grupa zaprta za operacijo grupe $G$.
\end{definicija}

\begin{opomba}[Preverjanje podgrupe]
    Da preverimo ali je $(Y, \circ)$ podgrupa $(X, \circ)$ je dovolj preveriti, da je $Y$ neprazna,
    da je $Y$ podmnožica $X$, in da velja $\forall a, b \in Y:\; a\circ b^{-1} \in Y$.
\end{opomba}

\begin{definicija}
    Pravimo, da je grupa $G$ abelova, če je operacija v njej še komutativna.
\end{definicija}

\begin{definicija}[Normalna podgrupa]\label{def:normalna_podgrupa}
    Podgrupa $H$ grupe $G$ je normalna podgrupa v $G$, če je invariantna 
    glede na konjugacijo, oziroma, če velja
    $gHg^{-1} = H$ za vsak $g \in G$. Normalnost grupe značimo z $H \triangleleft G$.
\end{definicija}

\begin{definicija}[Center grupe]\label{def:center_grupe}
    Definirajmo center grupe $G$ kot set vseh elementov, ki komutirajo z vsemi drugimi elementi grupe
    \[Z(G) = \{z \in G \mid \forall g\in G:\; zg = gz\}.\]
\end{definicija}

\begin{opomba}
    Center grupe $G$ je normalna podgrupa v $G$, saj za vsak $z \in Z(G)$ in $g, h \in G$ velja
    $gzg^{-1}h = gz g^{-1} h = g z h g^{-1} = g h z g^{-1} = h g z g^{-1} = h g g^{-1} z = hz$.
\end{opomba}

\begin{definicija}[Obseg]
Pravimo, da je $(G, \cdot, +)$ obseg, če je $(G \setminus \{0\}, \cdot)$ asociativna grupa, $(G, +)$ abelova grupa, 
in velja leva in desna distributivnost.
\end{definicija}

\subsubsection{Preslikave med grupami}

\begin{definicija}[Homeomorfizem grup]\label{def:homomorfizem_grup}
    Naj bosta $(G, \cdot)$ in $(H, *)$ dve grupi. Preslikava $\phi: G \to H$ je homomorfizem,
    če ohranja operacijo ($\phi(a \cdot b) = \phi(a) * \phi(b)\;\, \forall a,b\in G$).
\end{definicija}

\begin{definicija}[Izomorfizem grup]\label{def:izomorfizem_grup}
    Naj bosta $(G, \cdot)$ in $(H, *)$ dve grupi. Preslikava $\phi: G \to H$ je izomorfizem,
    če je bijektivna in je homeomorfizem, torej ohranja operacijo 
    ($\phi(a \cdot b) = \phi(a) * \phi(b)\;\, \forall a,b\in G$).

    Če takšna preslikava obstaja, pravimo, da sta grupi izomorfni, pišemo $G\cong H$.
\end{definicija}

\begin{definicija}[Avtomorfizem grupe]\label{def:avtomorfizem_grupe}
    Avtomorfizem grupe $G$ je izomorfizem grupe $G$ v samo vase. Drugače povedano je to 
    bijektivna preslikava $\phi: G\to G$, ki ohranja operacijo 
    ($\phi(a \cdot b) = \phi(a) \cdot \phi(b)\;\, \forall a,b\in G$).
\end{definicija}

\begin{definicija}[Grupa avtomorfizmov]\label{def:grupa_avtomorfizmov}
    Grupa avtomorfizmov grupe $G$ je grupa, katere elementi so 
    avtomorfizmi (\ref{def:avtomorfizem_grupe}) grupe $G$ pod operacijo kompozituma $\circ$. 
    Označimo jo z $\aut(G)$.
\end{definicija}

\begin{definicija}[Antiavtomorfizem]\label{def:antiavtomorfizem}
    Antiavtomorfizem grupe $G$ je preslikava iz grupe $G$ v grupo $G$, ki zamenja vrstni red množenja.
    Formalno je to bijektivna preslikava $\phi: G \to G$ taka, da 
    \[\forall g_1, g_2 \in G:\; \phi(g_1g_2) = \phi(g_2)\phi(g_1).\]
\end{definicija}

\begin{definicija}[Jedro homomorfizma]\label{def:jedro_homomorfizma}
    Naj bo $f: G \to H$ homomorfizem grup. Jedro homomorfizma $f$ je množica vseh elementov $g \in G$, 
    za katere velja $f(g) = e$, kjer je $e$ enota grupe $H$. Označimo ga z $\ker(f) = \{g \in G \mid f(g) = e\}$.
\end{definicija}


\begin{izrek}[Izrek o izomorfizmih]\label{iz:izomorfizmi}
Naj bosta $G$ in $H$ grupi in $f: G \to H$ homomorfizem. Potem veljajo 
\begin{enumerate}[label=(\roman*)]
    \item $\ker(f) = \{g \in G \mid f(g) = e\}$ je normalna podgrupa $G$,
    \item $\im(f) = \{f(g) \mid g \in G\}$ je podgrupa $H$,
    \item $f$ inducira izomorfizem $G / \ker(f) \cong \im(f)$.
\end{enumerate}  
\end{izrek}

\begin{dokaz}
    \begin{enumerate}[label=(\roman*)]
        \item Če sta $u, v\in \ker(f)$, potem $f(u v^{-1}) = f(u)f(v^{-1}) = f(u)f(v)^{-1} = e \cdot e^{-1} = e$.
        Torej $uv^{-1} \in \ker(f)$ in posledično je $\ker(f)$ podgrupa. Vzemimo zdaj poljubna 
        $g \in G$ in $h \in \ker(f)$, potem $f(ghg^{-1}) = f(g)f(h)f(g)^{-1} = f(g)ef(g)^{-1} = e$, torej 
        je $\ker(f)$ normalna podgrupa v $G$.
        \item Vzemimo poljubna $h_1, h_2 \in \im(f)$, to pomeni, da obstajata $g_1, g_2 \in G$, taka da
        $f(g_1) = h_1$ in $f(g_2) = h_2$. Potem $h_1 h_2^{-1} = f(g_1)f(g_2)^{-1} = f(g_1 g_2^{-1}) \in \im(f)$.
        \item Naj bo $K = \ker(f)$. Definiramo $\bar{f}$ kot $\bar{f}: G/K \to \im(f)$, $\bar{f}(gK) = f(g)$. 
        Ta preslikava je dobro definirana, saj če $gK = g'K$, potem $g^{-1}g' \in K$ in zato $f(g^{-1}g') = e$, 
        kar pomeni $f(g) = f(g')$. 
        Očitno je $\bar{f}$ homomorfizem, saj 
        $\bar{f}((g_1K)(g_2K)) = \bar{f}(g_1 g_2 K) = f(g_1 g_2) = f(g_1)f(g_2) = \bar{f}(g_1K)\bar{f}(g_2K)$, 
        in je tudi bijektivna, 
        saj je injektivna (ker $\ker(\bar{f}) = \{K\}$) in surjektivna (ker $\im(\bar{f}) = \im(f)$).
    \end{enumerate}
\end{dokaz}

\begin{definicija}[Delovanje grupe]\label{def:delovanje_grupe} % NUJNO
    Pravimo, da grupa $G$ deluje na množici $\Omega$, če obstaja preslikava
    $\cdot: G \times \Omega \to \Omega$, ki vsakemu paru $(g, v)$ priredi element $g\cdot v$
    tako, da velja:
    \begin{enumerate}[label=(\roman*)]
        \item $\forall g, h \in G, v \in \Omega:\; (gh) \cdot v = g \cdot (h \cdot v)$ (asociativnost delovanja),
        \item $\forall v \in \Omega:\; e \cdot v = v$ (identiteta).
    \end{enumerate}
    Iz teh pogojev sledi, da vsak element $g\in G$ določa permutacijo množice $\Omega$.
\end{definicija}

\begin{definicija}[Fiksni element]\label{def:fiksni_element} % NUJNA
    Naj bo $G$ grupa, ki deluje na $\Omega$ (\ref{def:delovanje_grupe}), in $g$ njen element. 
    Pravimo, da $g$ fiksira element $v\in \Omega$, če velja $g \cdot v = v$.
\end{definicija}

\begin{definicija}[Stabilizator]\label{def:stabilizator} % NUJNA.
    Za dano grupo $G$, ki deluje na množici $\Omega$ (\ref{def:delovanje_grupe}), pravimo, da je $G_v \leq G$ 
    stabilizator elementa $v \in \Omega$, če je to grupa vseh elementov grupe $G$, 
    ki fiksirajo $v$ (\ref{def:fiksni_element}), torej 
    $G_v = \{g \in G \mid g \cdot v = v\}$.

    Splošneje ga lahko definiramo tudi za poljubno podmnožico $V \subseteq \Omega$ kot 
    \[G_V = \{g \in G \mid g \cdot V = V\}.\]
\end{definicija}

\begin{definicija}[Orbita]\label{def:orbita}
    Orbita elementa $v \in \Omega$ prek grupe $G$ je podmnožica elementov iz $\Omega$, ki jih lahko dosežemo
    z delovanjem elementov iz $G$. Pišemo \[G \cdot v = \{g \cdot v \mid g \in G\}.\]

    Množici vseh orbit delovanja grupe $G$ na množici $\Omega$ rečemo 
    kvocientna množica $\Omega/G$, in jo pišemo kot $\Omega/G = \{G \cdot v \mid v \in \Omega\}$.
    Taka množica tvori particijo množice $\Omega$.
\end{definicija}

\begin{dokaz} %TODO: Could be polished.
    Pokažemo lahko, da so orbite ekvivalenčni razredi neke ekvivalenčne relacije,
    po tem bomo vedeli, da nujno razdelijo $X$ na disjunktne podmnožice, katerih unija je $X$.

    Definiramo relacijo $\sim$ na $X$ z 
    $x \sim y \iff \exists g \in G: g \cdot x = y \iff y \in G\cdot x$.
    Taka relacija je refleksivna $(x \sim x)$, saj je za enoto $e\in G$ $e \cdot x = x$. 
    Je simetrična $(x\sim y \implies y \sim x)$, saj če velja $x\sim y$,
    potem obstaja $g\in G$, tak da $g\cdot x = y$. Dobimo $x = g^{-1} \cdot y$.
    In je tranzitivna $(x\sim z \land z\sim y \implies x \sim y)$, saj če 
    velja $x\sim y$ (torej $y = g\cdot x$) in $z\sim y$ (torej $y = h \cdot z$), 
    potem je $h^{-1}g \cdot x = z$, kar pomeni, da je $x \sim z$.
\end{dokaz}

\begin{definicija}[Tranzitivno delovanje]\label{def:tranzitivno_delovanje}
    Pravimo, da grupa $G$ deluje tranzitivno na množici $\Omega$, če je z delovanjem elementov
    grupe $G$ mogoče poljuben element $v\in \Omega$ preslikati v poljuben drug element množice $\Omega$.
    Formalno je to natanko tedaj, ko ima vsak element $v\in \Omega$ orbito enako množici $\Omega$,
    oziroma ko ima delovanje grupe $G$ na množici $\Omega$ eno samo orbito ($|\Omega/G| = 1$).
\end{definicija}

\begin{definicija}[Koseti]\label{def:koseti}
    Množica kosetov podgrupe $H$ v grupi $G$ je množica $G/H = \{gH \mid g \in G\}$,
    sestavljena iz posameznih kosetov $gH = \{gh\mid h\in H\}$.

    Taka množica tvori particijo grupe $G$ na podmnožice enakih velikosti kot $H$.
\end{definicija}

\begin{dokaz}
    Ker je $H$ podgrupa v grupi $G$, vsebuje enoto $e$. Posledično je 
    $\bigcup_{g \in G} gH \supseteq \bigcup_{g \in G} ge = \bigcup_{g \in G} g = G$.

    Dalje imata $H$ in $gH$ enako število elementov, saj med njima obstaja bijekcija $h \mapsto gh$.
    
    Naj bo zdaj $x \in g_1 H \cap g_2 H$. Potem je $g_1h_1 = x = g_2 h_2$, torej $g_1 = g_2h_2h_1^{-1}$,
    oziroma $G_1 H \ni g_1 h = g_2h_2h_1^{-1}h \in g_2 H$ za poljuben $h\in H$. Tako smo pokazali
    $g_1 H \subseteq g_2 H$. Ker simetrično dobimo $g_2 H \subseteq g_1 H$, smo pokazali, da 
    sta si koseta bodisi enaka bodisi disjunktna.
\end{dokaz}

\begin{definicija}[Indeks podgrupe]\label{def:indeks_podgrupe}
    Indeks podgrupe $H$ v grupi $G$ je število
    kosetov $H$-ja v $G$, označujemo ga z $[G : H] = |\{gH\mid g\in G\}|$.
\end{definicija}

\begin{izrek}[Lagrangev izrek]\label{iz:lagrangev_izrek}
    Naj bodo $K \leq H \leq G$ grupe. Potem je red grupe $G$ deljiv z redom grupe $H$.
    Število kosetov podgrupe $H$ v $G$ je enako $[G : H] = |G| / |H|$ 
    in velja $[G : K] = [G : H][H : K]$.
\end{izrek}

\begin{dokaz}
    Po definiciji kosetov (\ref{def:koseti}) koseti tvorijo particijo grupe $G$ na 
    podmnožice velikosti $|H|$,
    torej očitno velja $[G : H] = |G| / |H|$. Posledično je tudi red grupe $G$ deljiv z redom grupe $H$.

    Zadnjo trditev pa dokažimo tako, da vzememo $S$ set predstavnikov kosetov za $K$ v $H$ in 
    $T$ set predstavnikov
    kosetov za $H$ v $G$. Potem je 
    $G = \bigsqcup_{t \in T} t H = \bigsqcup_{t \in T} t\bigsqcup_{s \in S} sK = \bigsqcup_{t \in T} \bigsqcup_{s \in S} tsK$.
    Tako je posledično set predstavnikov kosetov za $K$ v $G$ enak $TS = \{ts \mid t \in T, s \in S\}$, 
    kar pomeni, da je $[G : K] = |T||S| = [G : H][H : K]$.
\end{dokaz}

\begin{opomba}[Povezava med orbitami in stabilizatorji] % NE PREVEČ POMEMBEN.
    Če izberemo element $v \in \Omega$ in definiramo preslikavo $\phi: G \to X$ kot $g \mapsto g \cdot v$,
    potem je $\phi(G) = G \cdot v$ orbita elementa $v$ in 
    $\phi^{-1}(v) = G_v$ njegova stabilizatorska podgrupa.
\end{opomba}

\begin{opomba}[Bijekcija med koseti in orbitami]\label{op:bijekcija_koseti_orbita}
    Definiramo lahko preslikavo $\phi: G / G_x \to G\cdot x$ z $\phi(gG_x) = g \cdot x$. 
    Preslikava je dobro definirana, saj če vzamemo dva različna predstavnika $g$ in $h$ enakega koseta
    podgrupe $G_x$ v $G$, potem
    $gG_x = hG_x$, od koder sledi $h^{-1}g \in G_x$, 
    kar pomeni, da $h^{-1}g \cdot x = x$, oziroma, da res podata enak rezultat $g \cdot x = h \cdot x$.

    Preslikava $\phi$ je bijekcija. Pokažimo injektivnost tako, da vzamemo dva različna koseta $gG_x$ in $hG_x$
    in predpostavimo, da se slikata v enak element orbite $G\cdot x$, torej v $g\cdot x = h \cdot x$. 
    Potem je $h^{-1}g \cdot x = x$, kar pomeni, da $h^{-1}g \in G_x$, vendar potem sta koseta po definiciji enaka.

    Surjektivnost pa je očitna, saj je $\phi(G/G_x) = \phi(\{gG_x\mid g\in G\}) = \{g\cdot x \mid g\in G\} = G\cdot x$.

    Posledično je število kosetov podgrupe $G_x$ v $G$ enako številu elementov orbite $G\cdot x$,
    oziroma, da je $[G : G_x] = |G\cdot x|$.
\end{opomba}

\begin{opomba}[Računanje reda grupe z orbitami in stabilizatorji]\label{op:orbita_stabilizator_red}
    Zdaj lahko izračunamo red grupe $G$ s pomočjo orbit in stabilizatorjev.
    Po Lagrangevem izreku (\ref{iz:lagrangev_izrek}) je $|G| = |G_x|[G : G_x]$.
    Dalje pa z opombo \ref{op:bijekcija_koseti_orbita} dobimo $|G| = |G_x||G\cdot x|$. 
\end{opomba}

    S pomočjo rezultata iz opombe \ref{op:orbita_stabilizator_red} lahko izračunamo število avtomorfizmov kocke $\mathcal{K}$.
    Kocka ima $8$ oglišč, $12$ robov in $6$ ploskev. Fiskirajmo eno vozlišče $v$, tedaj imamo
    $|\aut(\mathcal{K})| = |\aut(\mathcal{K})_v||\aut(\mathcal{K})\cdot v|$. Ker se lahko prepričamo, da grupa
    avtomorfizmov deluje tranzitivno na množici oglišč, je $|\aut(\mathcal{K})\cdot v| = 8$. 

    Rezultat opombe pa lahko uporabimo še enkrat, in zdaj sicer na stabilizatorsko grupo $\aut(\mathcal{K})_v$, 
    ki fiksira vozlišče $v$. Fiksiramo še $u \in N_1(v)$ in dobimo 
    $|\aut(\mathcal{K})_v| = |\aut(\mathcal{K})_{v, u}||\aut(\mathcal{K})_v \cdot u|$.
    Ker je $u$ sosed vozlišča $v$, ga lahko avtomorfizmi, ki fiksirajo $v$, preslikajo zgolj v enega izmed
    $3$ sosedov vozlišča $v$, kar pomeni, da je $|\aut(\mathcal{K})_v \cdot u| = 3$.

    Da bi izračunali še $|\aut(\mathcal{K})_{v, u}|$, iskiramo še eno vozlišče $w \in N_1(v)$.
    Ker je $u$ v $\aut(\mathcal{K}_{v, u})$ fiksiran, lahko $w$ preslikamo zgolj v enega izmed $2$ sosedov vozlišča $v$,
    tako je $|\aut(\mathcal{K})_{v, u} \cdot w| = 2$. 
    
    Na koncu pa lahko fiksiramo še zadnjega soseda vozlišča $v$, ki je $z\in N_1(v)$, 
    in ker so vsi ostali sosedi vozlišča $v$ že fiksirani dobimo $|\aut(\mathcal{K})_{v, u, w} \cdot z| = 1$.

    Seveda pa en trikrak tudi natanko določa kocko in zato je $|\aut(\mathcal{K})_{v, u, w}| = 1$.
    Tako smo pokazali, da je $|\aut(\mathcal{K})| = 8 \cdot 3 \cdot 2 \cdot 1 = 48$.
\end{zgled}

Po opombi \ref{op:orbita_stabilizator_red} je red avtomorfizmov Petersenovega grafa enak $120 \cdot 1 = 120$.
\end{zgled}

\end{opomba}

\section{Mooreovi grafi}
% definicija, slike
Mooreovi grafi so ekstremalni grafi, ki glede na dan premer ter dano 
maksimalno stopnjo dosežejo 
najbolj optimistično zgornjo mejo za število vozlišč.
Leta 1960 sta jih tako 
poimenovala Hoffman in Singelton po Edwardu F. Mooru, ki je prvi zastavil vprašanje o 
opisu in klasifikaciji teh struktur.

\begin{definicija}\label{def:moore}
Graf $G$, s premerom $D$ in maksimalno stopnjo $\Delta$, 
je Mooreov, če dosega najbolj optimistično 
zgornjo mejo za število vozlišč -- Mooreovo mejo, podano z izrazom:
\begin{equation}\label{eq:moorebound}
    1 + \Delta + \Delta (\Delta - 1) + \ldots + \Delta (\Delta - 1)^{D - 1}.
\end{equation}
Od tu dalje bomo simbola $D$ in $\Delta$ vselej uporabljali za označevanje parametrov 
Mooreovega grafa.
\end{definicija}

\begin{opomba}\label{op:mooreova_meja}
Kakor smo rekli, izrazu~\ref{eq:moorebound} pravimo Mooreova meja. Pomen take meje postane jasen, 
če opazujemo strukturo grafa iz izbranega vozlišča $v \in V$. Tako vidimo, da je
na razdalji $1$ od $v$ največ $\Delta$ sosedov. Na vsaki naslednji razdalji $i$ 
(kjer $1 < i < D)$ pa ima vsako vozlišče na voljo kvečjemu $\Delta - 1$ sosedov,
ki v prejšnjih korakih še niso bili obiskani.
Mooreov graf je tako tisti mejni primer, kjer se veje sosedstva nikoli ne prekrivajo 
dokler ne dosežemo premera $D$. Ekvivalentno bi lahko rekli tudi, da je vsak Mooreov
graf v resnici neko drevo, katerega listi so med seboj dodatno
povezani ravno tako prepleteno, da grafu res omogočijo željen premer, a da hkrati njihove
stopnje ne presegajo maksimalne stopnje $\Delta$.
Slednji pogled na strukturo Mooreovega grafa je prikazan na sliki
(\ref{fig:mooretree}).
\end{opomba} % TODO!! ZAkaj sliko narobe labela???!?!

\subsection{Lastnosti Mooreovih grafov}

\begin{trditev}\label{tr:mooreregularpaths}
Vsak Mooreov graf premera $D$ in stopnje $\Delta$ je regularen z ožino ${2D+1}$.
Velja tudi to, da med poljubnima dvema njegovima vozliščema 
obstaja natanko ena najkrajša pot dolžine največ $D$.
\end{trditev}




\begin{dokaz}
Če Mooreov graf $G$ ne bi bil regularen, 
bi obstajalo neko vozlišče $u$, ki bi imelo stopnjo manjšo od $\Delta$. 
Vendar če bi konstrukcijo najbolj optimistične zgornje meje (\ref{op:mooreova_meja}) za število vozlišč
začeli iz takega vozlišča, potem bi videli, da bi naša meja 
$1 + \sum_{i=1}^{D-1} \deg(u)(\Delta - 1)^{i-1}$
vedno bila strogo manjša od 
Mooreove. Graf $G$ tako po definiciji ne bi mogel biti Mooreov.

Prav tako smo v opombi \ref{op:mooreova_meja} videli, da je struktura Mooreovega grafa enaka "drevesu", katerega 
zgolj listi so med seboj dodatno povezani. Če odmislimo te dodatne povezave, potem je jasno, 
da takšno "drevo"{} tvori ravno drevo enoličnih najkrajših poti dolžine največ $D$
-- od koder zadnji del trditve sledi. Hkrati pa se z zavedanjem takih 
dodatnih povezav med listi prepričamo tudi o njegovi ožini.
\end{dokaz}



V nadaljevanju bomo poskušali razkriti nekaj več o njihovih lastnostih, jih poiskati, ter v nekaterih 
primerih utemeljili tudi njihov neobstoj.


\subsubsection{Matrična enačba Mooreovih grafov}

%citiraj izpeljavo iz uvoda v teorijo
Naj bo $A$ sosednostna matrika Mooreovega grafa stopnje $\Delta$. 
Definirajmo matriko $G^{\Delta, i}(A)$ 
tako, da njeni elementi $G^{\Delta, i}_{uv}(A)$ podajo število poti  
dolžine največ $i$ med $u$ in $v$. Ker so edine poti dolžine $0$ vozlišča sama, 
velja $G^{\Delta, 0} = I$; za poti dolžine največ $1$ pa $G^{\Delta, 0} = A + I$.
Produkt $AG^{\Delta, i-1}$ prešteva sprehode dolžine največ $i$, ki jih dobimo tako, 
da potem dolžine $(i-1)$ dodamo še en korak. Pri tem upoštevamo, da se nekateri
sprehodi nadaljujejo, raje kot naprej, nazaj v nivo $(i-2)$.
Ker je Mooreov graf $\Delta$~-regularen (\ref{tr:mooreregularpaths}), za vsak par vozlišč $(u, v)$ obstaja natanko
$(\Delta - 1)G^{\Delta, i-2}_{uv}(A)$ takšnih sprehodov, ki niso poti in jih je treba 
odšteti. S tem razmislekom smo, ob upoštevanju trditve \ref{tr:mooreregularpaths}, ravnokar dokazali naslednjo trditev.

\begin{trditev}\label{tr:moorematrix}
Sosednostna matrika Mooreovega grafa $A$ zadošča matrični enačbi 
\[G^{\Delta, D}(A) = J,\]
kjer je 
$J$ matrika samih enic, $G^{\Delta, D}$ pa dobljena rekurzivno z:
\begin{equation}
\begin{cases} 
G^{\Delta, 0}(A) = I \\
G^{\Delta, 1}(A) = A + I \\
G^{\Delta, i}(A) = A G^{\Delta, i-1}(A) - (\Delta-1) G^{\Delta, i-2}(A) & \text{za } 2 \leq i \leq D
\end{cases}
\end{equation}
\end{trditev}

\begin{opomba}
    Velja celo to, da regularni graf z defektom $\delta$ zadošča enačbi
    $G^{\Delta, D} = J + B$, kjer je $B$ matrika z 
    stolpičnimi in vrstičnimi seštevki enaki $\delta$ \cite[str. 15]{MillerSiran2013}.
    Več o grafih z defektom bomo povedali kasneje~(\ref{sec:mooreovi_grafi_z_defektom}).
\end{opomba}

\begin{opomba}\label{tr:moore_matrix_solution}
Rekurzivna enačba matrike $G^{\Delta, i}(A)$ je navadna diferenčna enačba 
drugega reda s konstantnimi koeficienti.
Posledično je njena rešitev oblike 
\[G^{\Delta, D}(A) = \alpha \lambda_1(A)^D + \beta \lambda_2(A)^D,\] 
kjer sta $\lambda_1(A) = \frac{A + \sqrt{A^2 - 4(\Delta-1)}}{2}$ in 
$\lambda_2(A) = \frac{A - \sqrt{A^2 - 4(\Delta-1)}}{2}$ ničli karakteristične enačbe.
Slednje se poenostavi v:
\begin{equation}
    G^{\Delta, D}(A) = \frac{s^D}{\sin(\phi)}[\sin((D+1)\phi) + \frac{1}{s}\sin(D\phi)],
\end{equation}
kjer je $s = 2\sqrt{\Delta-1}$ in $\phi=\arccos(\frac{A}{2\sqrt{\Delta-1}})$.
\end{opomba}

\begin{opomba}
Matrika samih enic $J$ ima lastno vrednost $n$, enako Mooreovi meji, za lastni vektor $\mathbf{1}$, 
kjer je $n$ njena dimenzija. Ni se težko prepričati, da je lastna vrednost
matrike sosednosti $A$ za lastni vektor $\mathbf{1}$ enaka ravno $\Delta$,
namreč graf je $\Delta$-regularen. Rečemo lahko, da je $\Delta$ trivialna
lastna vrednost matrike $A$. Ker velja $G^{\Delta, D}(A)\cdot\mathbf{1} = J\cdot\mathbf{1}$,
velja tudi, da je vrednost $G^{\Delta, D}(\Delta)$ enaka natanko Mooreovi meji.

Ostale lastne vrednosti matrike $J$ so enake $0$.
Enačbo $G^{\Delta, D}(A) = J$ lahko tako pomnožimo z lastnim vektorjem $v$
matrike $J$ za lastno vrednost $0$. Tako dobimo $G^{\Delta, D}(A)v = 0$,
oziroma $ G^{\Delta, D}(\lambda) = 0$ za vsako netrivialno lastno 
vrednost $\lambda$ matrike $A$.

Posledično vemo, da je $(x - \Delta)G^{\Delta, D}(x)$ karakteristični 
polinom matrike sosednosti Mooreovega grafa.
\end{opomba}

\subsubsection{Izrek Hoffman-Singeltona}
Naj bo premer $D$ enak $2$, ter stopnja $\Delta$ večja od $1$. Tedaj za matriko 
sosednosti Mooreovega grafa (po trditvi \ref{tr:moorematrix}) velja:
\begin{equation}\label{eq:moorematrix2}
A^2 + A - (\Delta-1)I = J,
\end{equation}
kjer je $I$ identiteta in $J$ matrika samih enic.
Iz definicije~\ref{def:moore} je razvidno, da je red takega Mooreovega grafa enak
$\Delta^2 + 1$. Velja tudi, da je Mooreov graf premera $2$ krepko regularen 
s parametri $(\Delta^2 + 1, \Delta, 0, 1)$, kar pomeni, da je vsak par sosednjih vozlišč 
povezan z $0$ skupnimi sosedi (glej omejitev ožine~\ref{tr:mooreregularpaths}), 
vsak par nesosednjih vozlišč pa je povezan z $1$ skupnim sosedom (glej omejitev premera~\ref{def:moore}).

Matrika $J$ je dimenzije $(\Delta^2 + 1) \times (\Delta^2 + 1)$.
Ker velja $J\cdot\mathbf{1} = (\Delta^2 + 1)\cdot\mathbf{1}$, je $(\Delta^2 + 1)$ njena 
lastna vrednost in $\mathbf{v}_1 = \mathbf{1}$ njen lastni vektor. 
Ker je $\rank(J) = 1$, in s tem $\dim(\ker(J)) = \Delta^2$ \cite{enwiki1344320194}, je njena večkratnost enaka $1$.
Vrednost vseh ostalih lastnih vrednosti je posledično enaka
$0$, z večkratnostjo $\Delta^2$,
za lastne vektorje 
$(\mathbf{v}_2, \ldots, \mathbf{v}_{\Delta^2}) = (\mathbf{e}_1 - \mathbf{e}_2, \ldots, \mathbf{e}_1 - \mathbf{e}_{\Delta^2 + 1})$,
kjer je $\mathbf{e}_i$ $i$-ti standardni bazni vektor.

S tem ugotovitvam, z množenjem matrične enačbe~\ref{eq:moorematrix2} z 
lastnimi vektorji $\mathbf{v}_i$ dobimo naslednji dve enačbi:

\begin{align}
x^2 + x - (\Delta - 1) = \Delta^2 + 1 \label{eq:karakteri1}\\
x^2 + x - (\Delta - 1) = 0 \label{eq:karakteri2}
\end{align}


Iz enačbe~\ref{eq:karakteri1} dobimo lastno vrednost $\Delta$ z večkratnostjo $1$, kakor pričakovano,
iz enačbe~\ref{eq:karakteri2} pa dobimo dve lastni 
vrednosti $r$ in $s$ enaki $(-1 \pm \sqrt{4\Delta - 3})/2$, s pripadajočima večkratnostima $a$ in $b$, 
kjer $1 + a + b = 1 + \Delta^2$.
Ker je sled matrike sosednosti vedno enaka $0$, in ker velja 
$\sled(A) = \sum_{\lambda \text{ je lastna vr.}} \lambda$,
je $\Delta + a r + b s = 0$.

Če sta $r$ in $s$ iracionalni števili je $a = b$, potem imamo
$\Delta + a r + b s = 0 \implies \Delta - \Delta^2 / 2 = 0 \implies \Delta = 2$.
Sicer pa sta $r$ in $s$ celi števili, namreč $4\Delta - 3$ je vedno liho število, posledično 
je tudi njegov koren vedno lih. Tako nastavimo $4\Delta - 3 = m^2$ za nek $m \in \ZZ_+$, 
oziroma $\Delta = \frac{m^2 + 3}{4}$, od koder dobimo: 
$\Delta + a r + b s = 0 \implies m^5 + m^4 + 6m^3 - 2m^2 + 9m - 15 = 0$.
Ker ima slednji moničen polinom celoštevilske koeficiente, so po izreku o 
celoštevilskih ničlah~\cite{enwiki1346919033}, njegove rešitve deljitelji števila $15$,
torej $m \in \{1, 3, 5, 15\}$, kar nam da možne rešitve za $\Delta$ enake $\{1, 3, 7, 57\}$.

Iz zgornjih dveh enačb, ki sta linearni v $a$ in $b$, izvemo tudi, da sta $a$ in $b$ enaka 
$\frac{1}{2} [\Delta^2 \mp \frac{2\Delta - \Delta^2}{\sqrt{4\Delta - 3}}]$.

\begin{izrek}[Hoffman-Singeltonov izrek]\label{iz:hoffmansingeltonizrek}
    Mooreovi grafi z premerom $D = 2$ in stopnjo $\Delta > 1$ obstajajo le za $\Delta \in \{2, 3, 7, 57\}$.
    Njihove lastne vrednosti so $\Delta$ z večkratnostjo $1$, 
    $\frac{1}{2}(-1 + \sqrt{4\Delta - 3})$ z večkratnostjo 
    $\frac{1}{2} [\Delta^2 - \frac{2\Delta - \Delta^2}{\sqrt{4\Delta - 3}}]$, ter 
    $\frac{1}{2}(-1 - \sqrt{4\Delta - 3})$ z večkratnostjo
    $\frac{1}{2} [\Delta^2 + \frac{2\Delta - \Delta^2}{\sqrt{4\Delta - 3}}]$.
\end{izrek}

\begin{opomba}
Izrek smo že dokazali, vendar pa bi do enakega 
lahko prišli tudi z uporabo dejstva, 
da so Mooreovi grafi premera $2$ krepko regularni s parametri
$(n, k, \lambda, \mu) = (\Delta^2 + 1, \Delta, 0, 1)$. Namreč
krepko regularni grafi zadoščajo matrični enačbi $A^2 = kI + \lambda A + \mu(J - I - A)$,
ter spekter matrike $A$ je znan~\cite{enwiki1342817669}.
\end{opomba}


\subsection{Obstoj Mooreovih grafov}

Hitro lahko sestavimo nekaj trivialnih Mooreovih grafov, namreč, če je premer $D=1$, potem 
je Mooreov graf poln graf $K_{\Delta + 1}$ za vsako stopnjo $\Delta$, če je stopnja 
$\Delta = 1$, potem je Mooreov graf ciklični graf $C_{2D + 1}$ za vsak premer $D$.

Hoffman-Singeltonov izrek~\ref{iz:hoffmansingeltonizrek} pa nam pove, da za premer $D=2$ obstajajo 
največ štirje Mooreovi grafi, in sicer za stopnje $\Delta \in \{2, 3, 7, 57\}$.
Izkaže se, da so prvi trije Mooreovi grafi unikatni in znani, imenovani Ciklični graf $C_5$,
Petersenov graf, ter Hoffman-Singeltonov graf, medtem ko obstoj grafa valence $57$ ostaja odprt problem.

Predstavimo sedaj netrivialne primere podrobneje, v naslednjih nekaj sekcijah.

\subsubsection{Petersenov graf}\label{sec:petersenov_graf}

Konstrukcija Petersenovega grafa je preprosta, in bi lahko bila puščena bralcu kot vaja, vendar jo bomo vseeno 
predstavili, in sicer na več različnih načinov, saj nam bo služila kot dober uvod za razumevanje 
kasneje kompleksnejših konstrukcij.

Začnimo s konstrukcijo, ki izhaja iz drevesnega grafa~(glej \ref{def:moore}). 
Vzemimo neko začetno vozlišče $v$, ter ga povežimo z $\Delta=3$ sosedi, ki jih poimenujemo $v_1, v_2, v_3$.
Vsakega izmed teh povežimo še z dvema novima vozliščema, ki jih poimenujemo 
$v_{11}, v_{12}, v_{21}, v_{22}, v_{31}, v_{32}$. V tem trenutku v rokah držimo drevesni graf. Sedaj moramo
njegove liste povezati tako, da ne ustvarimo ciklov dolžine $3$ ali $4$~(glej \ref{tr:mooreregularpaths}). 
Brez škode za splošnost lahko 
povežemo $v_{11}$ z $v_{21}$ ter z $v_{31}$, $v_{12}$, po tem smo prisiljeni povezati $v_{12}$ z 
$v_{22}$ in $v_{32}$, brez drugih izbir tako ostanemo do konca konstrukcije, povežemo $v_{21}$ z
$v_{32}$, ter $v_{22}$ z $v_{31}$.

cikel $C_5$. Začnimo našo konstrukcijo zato z ciklom $C_5$. Ker vsakemu vozlišču v tem ciklu manjka še en sosed, mu ga dodamo.
Tu opazimo, da vozlišča cikla $C_5$ ne smejo imeti skupnih sosedov (izven tega $C_5$). Če se sedaj osredotočimo
na eno izmed novo dodanih vozlišč, vidimo, da je lahko povezano le z dvema vozliščema, ki sta soseda nasprotnih dveh 
vozlišč v ciklu $C_5$. Hkrati mora biti povezano z natanko dvema. Povežemo ga z njima. Simetrično nato storimo 
za druge. Taki konstrukciji
rečemo konstrukcija iz petkotnika, poslužili se je bomo tudi pri konstrukciji Hoffman-Singeltonovega grafa.

Petersenov graf lahko konstruiramo tudi s pomočjo množice $X = \{1, 2, 3, 4, 5\}$, 
kjer so vozlišča grafa predstavljena z $2$-elementnimi podmnožicami množice $X$, medtem ko so povezave
med vozlišči predstavljene z njihovo disjunktnostjo. Pišemo lahko 
$V(\text{Petersenov graf}) = \{\{a, b\}\mid\{a, b\}\subset S_5\}$ ter 
$uv \in E(\text{Petersenov graf}) \iff u \cap v = \emptyset$. 
Ta konstrukcija res poda Mooreov graf na $10$ vozliščih, stopnje $3$ in premera $2$, 
saj ima ustvarjen graf $\binom{5}{2} = 10$ vozlišč, 
za poljubno podmnožico $\{a, b\}$ pa obstajajo natanko $\binom{5-2}{2} = 3$ disjunktni pari,
ter ker so cikli krajši od $5$, po konstrukciji, nemogoči.
Dalje vidimo, da vsaka permutacija $\sigma \in S_5$ simetrične grupe na petih elementih, deluje na vozliščih
kot avtomorfizem (ohranja sosedstvo). Ker je po zgledu~\ref{todo:TODO} $\aut(\text{Petersenov graf}) = 5!$,
in ker $S_5 \leq \aut(\text{Petersenov graf})$, smo dobili tudi, da je $\aut(\text{Petersenov graf}) \cong S_5$.

Ker lahko z delovanjem simetrične grupe $S_5$ na množici vozlišč, poljubno vozlišče preslikamo v katerokoli
drugo, je Petersenov graf tudi vozliščno tranzitiven.



\subsubsection{Hoffman-Singeltonov graf}

Naj bo $H$ graf, ki ga iščemo, torej Mooreov graf premera $2$ ter stopnje $7$.
Tak graf sestoji iz $7\cdot 7 + 1 = 50$ vozlišč (\ref{def:moore}) in ima 
ožino $5$ (\ref{tr:mooreregularpaths}).
Po korakih Conga Fana in Allena J. Schwenka \cite{fan1993structure} dokažimo,
da tak graf, ki mu pravimo Hoffma-Singeltonov graf, obstaja in je enoličen.

Ker ima graf ožino $5$ v njem najdemo vsaj en cikel $C_5$. Označimo ga s
$Q=v_1 v_2 v_3 v_4 v_5 v_1$ in mu recimo centralni cikel. 
V nadaljevanju bomo opazovali njegove sosede
in s tem gradili strukturo grafa $H$. Za ta namen začnimo z nekaj naslednjih definicij, katerim 
bo nato sledila serija trditev in lem, ki nas bodo pripeljale do iskanega grafa. Dodatno 
bomo ob koncu tudi prepričani o njegovi unikatnosti in številu njegovih avtomorfizmov.

\begin{definicija}\label{def:gruce}
    Naj bo $Q = v_1 v_2 v_3 v_4 v_5 v_1$ centralni cikel grafa $H$. 
    Definirajmo množico $N_1(Q)$ kot množico vseh vozlišč, 
    ki so sosedi vsaj enega vozlišča iz $Q$, vendar ne ležijo v $Q$.
    Potem vsakemu vozlišču $v_i$ centralnega cikla dodelimo gručo $\gr_i$, ki vsebuje vsa
    njegova sosedna vozlišča, katera niso del $Q$.
    \begin{equation}\label{eq:gruce_definicija}
        \forall v_i \in Q: \; \gr_i = \{\, u \in V(H) \setminus Q \mid u \sim v_i\,\} = N_1(v_i) \setminus V(Q).
    \end{equation}
    Pravimo, da je vozlišče $v_i\in Q$ center svoje gruče $\gr_i$.
\end{definicija}

\begin{opomba}
    Seveda iz definicije neposredno sledi, da njihova unija pokrije celotno množico $N_1(Q)$,
    in da, zaradi naslednji trditve (\ref{tr:gruce_preparation}), vsaka gruča vsebuje natanko $5$ vozlišč
\end{opomba}

\begin{trditev}\label{tr:gruce_preparation}
    Poljubno vozlišče centralnega cikla $v \in Q$ ima natanko $7 - 2 = 5$ sosedov nevsebovanih v $Q$.
    Tako vsaka gruča vsebuje natanko $5$ vozlišč.
\end{trditev}
\begin{dokaz}
    Če to ne bi veljalo, bi med vozlišči $Q$ v grafu $H$ našli cikla $C_3$ in $C_4$, kar bi 
    bilo v nasprotju s predpostavko o ožini (\ref{tr:mooreregularpaths}).
\end{dokaz}

\begin{trditev}\label{tr:lastnosti_gruc}
Gruče, kot definirane zgoraj, v grafu $H$ zadoščajo naslednjim lastnostim:
\begin{enumerate}[label=(\roman*)]
\item Gruče so si paroma disjunkne: \(\forall i, j: \;\gr_i \cap \gr_j = \emptyset\),
    \label{itm:g1}
\item vozlišča znotraj posamezne gruče so nepovezana: 
\(\forall i. \forall u, v \in \gr_i: \; u \not\sim v\),
    \label{itm:g2}
\item med gručama $\gr_i$ in $\gr_j$ ni povezave, če sta njuna centra \(v_i\) in \(v_j\) sosedna v \(Q\),
sicer pa je vsako vozlišče gruče \(\gr_i\) sosedno največ enemu vozlišču 
gruče \(\gr_j\): \\ \(\forall i, j. \forall u\in \gr_i:\; \left( (v_i \sim v_j 
\implies \gr_i \not\sim \gr_j) \lor  (|N_1(u) \cap \gr_j| \leq 1)\right)\)
    \label{itm:g3}
\end{enumerate}
\end{trditev}

\begin{dokaz}
Vzemimo vozlišči $u, w \in \gr_i$, ter vozlišče $z \in \gr_j$, kjer $i \neq j$.
Pri dokazih uporabimo metodo protislovja, kjer se dokaz zaključi ko najdemo 
vsaj en cikel dolžine manj kot 5. To bi namreč bilo v nasprotju z ožino grafa (\ref{tr:mooreregularpaths}).
\begin{enumerate}[label=(\roman*)]
\item Predpostavimo, da obstajata dve gruči, katerih presek ni prazen.
Brez škode za splošnost predpostavimo $u = z \in \gr_i \cap \gr_j$.
Tedaj v grafu $H$ obstaja cikel $u v_i \ldots v_j u$, ki je bodisi $C_3$ ali $C_4$.
\item Naj obstaja gruča $\gr_i$ ter znotraj nje vozlišči $u$ ter $w$, ki sta povezani.
Tedaj v grafu obstaja cikel $C_3 = u v_i w u$.
\item Naj bosta najprej centra gruč $v_i$ ter $v_j$ sosedna, ter $u$ in $z$
povezana. Tedaj v grafu $H$ obstaja cikel $C_4 = u v_i v_j z u$.
Predpostavimo sedaj, da $v_i \not \sim v_j$, ter da je vozlišče $z$ povezano 
tako z $u$ kot z $w$. Tedaj obstaja cikel $C_4 = z u v_i w z$.
\end{enumerate}
\end{dokaz}

\begin{posledica}\label{po:enolicna_povezava_iz_nq_v_q}
    Poljubno vozlišče $u\in N_1(Q)$ je povezano z natanko enim vozliščem iz centralnega
    cikla $Q$. Namreč po definiciji gruč (\ref{def:gruce})in trditvi o disjunktnosti gruč 
    (\ref{tr:lastnosti_gruc}\ref{itm:g1}), je $u$ del natanko ene gruče, 
    in s tem povezan zgolj z njenim centrom (glej sliko \ref{fig:hoffman_1}).
\end{posledica}

\begin{posledica}\label{po:lastnosti_gruc}
Stopnja poljubnega vozlišča v podgrafu $H[N_1(Q)]$ induciranem z
množico sosedov $Q$, je enaka največ $2$. 
Po točki~\ref{itm:g3} trditve~\ref{tr:lastnosti_gruc} namreč velja, da je vsako vozlišče dane 
gruče $\gr_i$ lahko povezano zgolj z gručami katerih centri si niso sosedni z $v_i$.
Taki gruči sta dve, ter do vsake izmed njiju lahko vodi največ ena povezava (\ref{tr:lastnosti_gruc}\ref{itm:g3}).
\end{posledica}

\begin{opomba}\label{op:hf_n2}
Ogledali smo si strukturo centralnega cikla $Q$ ter strukturo njegovih sosedov $N_1(Q)$,
katerih je, mimogrede, natanko $5\cdot 5 = 25$ 
(glej \ref{tr:gruce_preparation} in \ref{tr:lastnosti_gruc}\ref{itm:g1}).

Da bomo o strukturi grafa $H$ lahko povedali še kaj več, si oglejmo še preostala vozlišča, 
katerih je $50 - 5 - 25 = 20$. Ker so le ta, na razdalji natanko $2$ od vozlišč centralnega cikla $Q$,
jih prikladno označimo z $N_2(Q)$.
\end{opomba}

\begin{trditev}\label{tr:st_povezav_iz_n2q}
    Naj bo $q$ vozlišče iz množice $N_2(Q)$ (\ref{op:hf_n2}), tedaj ima 
    lahko z vozlišči iz $N_1(Q)$ skupno največ $5$ povezav.
\end{trditev}
\begin{dokaz}
    Če bi jih imelo več, bi to pomenilo, da je $q$ povezano z neko gručo $\gr_i$ dvakrat,
    recimo z $u$ in $w$. Potem bi v grafu $H$ obstajal cikel $C_4 = q u v_i w q$,
    kar bi bilo v nasprotju z ožino grafa (\ref{tr:mooreregularpaths}).
\end{dokaz}

\begin{trditev}\label{tr:st_sosedov_v_n2q}
Vsako vozlišče $u$, ki pripada gruči $\gr_i$, ima natanko $4$ sosede v $N_2(Q)$. Poleg tega
vozlišča iz iste gruče nimajo skupnih sosedov v $N_2(Q)$.
\[
\forall i:\; \forall u, w \in \gr_i,\,u \neq w:\ |N_1(u) \cap N_2(Q)| = 4 \ \land \ (N_1(u) \cap N_1(w)) \cap N_2(Q) = \emptyset,
\]
\end{trditev}

\begin{dokaz}
Kot v dokazu trditve~\ref{tr:lastnosti_gruc} bomo tudi tu vzeli vozlišči $u, w \in \gr_i$,
ter vozlišče $z\in\gr_j$, kjer $i \neq j$. Tudi tu bomo uporabili metodo protislovja.

Opazujmo najprej vozlišče $w$. Tako vozlišče ima vsaj 4 sosede v $N_2(Q)$, 
saj je njegova stopnja enaka $7$, povezan pa je z vozliščem 
$v_i \in Q$ ter po posledici~\ref{po:lastnosti_gruc} še z največ dvema
vozliščema iz $N_1(Q)$. Hkrati pa v $N_2(Q)$ nima več kot 4 sosede, kar pokažemo z enostavnejšim 
protislovjem. Predpostavimo lahko nasprotno, potem bi iz množice vozlišč $N(Q)$ 
v množico vozlišč $N_2(Q)$ vodilo vsaj 
$24\cdot 4 + 1\cdot 5 = 101$ povezav, kar je več kot $20\cdot 5 = 100$, ki bi vodilo v obratno 
smer. Namreč vozlišče iz $N_2(Q)$ ima lahko največ $5$ sosedov v $N(Q)$ 
(\ref{tr:st_povezav_iz_n2q}).
\end{dokaz}

\begin{posledica}\label{po:regularnosti_podgrafov}
    Podgrafa $H[N_1(Q)]$ in $H[N_2(Q)]$ sta oba 2-regularna. 
    %Posledično ima vsako vozlišče iz $N_2(Q)$ natanko 5 sosedov v $N(Q)$.
\end{posledica}

\begin{dokaz}
    Vozlišče iz podgrafa $H[N_1(Q)]$ je po trditvi~\ref{tr:st_sosedov_v_n2q} povezano 
    z natanko 4 vozlišči iz $N_2(Q)$, torej iz množice $N_1(Q)$ v množico $N_2(Q)$
    vodi natanko $25\cdot 4 = 100$ povezav. V dokazu ravno te trditve pa smo pokazali,
    da lahko v nasprotno smer vodi toliko povezav zgolj, če ima vsako vozlišče iz $N_2(Q)$ 
    natanko $5$ sosedov v $N_1(Q)$, od koder naravno sledi, da je $H[N_2(Q)]$ $2$-regularen.
    
    Poleg tega pa je, po 
    posledici~\ref{po:enolicna_povezava_iz_nq_v_q}, vsako vozlišče iz $H[N_1(Q)]$ povezano z 
    natanko enim vozliščem iz centralnega cikla $Q$. Skupno ima tako vozlišče torej natanko $5$ povezav izven 
    $H[N_1(Q)]$ in zato točno $2$ povezavi znotraj $H[N_1(Q)]$, torej je res $2$-regularen.
\end{dokaz}

\begin{trditev}\label{tr:unija_ciklov}
    Podgraf $H[N_1(Q)]$ je disjunktna unija ciklov, 
    katerih dolžina je večkratnik števila $5$.
\end{trditev}

\begin{dokaz}
    Tu se preprosto skličemo na posledico~\ref{po:regularnosti_podgrafov}, ki nam pove,
    da je $H[N_1(Q)]$ 2-regularen, kar je ekvivalentno temu, da je disjunktna unija ciklov.
    Lastnosti gruč~\ref{tr:lastnosti_gruc}\ref{itm:g3} nam nato dodatno povedo, da lahko vsak 
    cikel zaključimo šele v koraku, ki je večkratnik števila $5$. 
\end{dokaz}

Sedaj se osredotočimo na spektralno analizo grafa $H$ s pomočjo katere bomo 
lahko sklepali več o številu in velikosti ciklov, ki sestavljajo podgraf $H[N(Q)]$. 
Iz lastnosti Mooreovih 
grafov~\ref{iz:hoffmansingeltonizrek} vemo, da je njegov spekter 
enak $\{7^{(1)}, 2^{(28)}, (-3)^{(21)}\}$.

Pred nadaljevanjem je smiselno predstaviti lemo, ki govori o lastnih vrednostih cikličnih
grafov ter splošno znan izrek, ki govori o lastnih vrednostih podgrafov.

\begin{lema}\label{le:cikellastne}
Naj bo dan cikel $C_n$ ter njegova sosednostna matrika $A$. Tedaj so lastne vrednosti 
matrike $A$ enake $2\cos(2\pi k/n)$, za vsak $k \in \{0, 1, \ldots, n-1\}$. Posebno 
ima vsak cikel natanko eno lastno vrednost enako $2$.
\end{lema}
\begin{dokaz}
    Naj bo $A$ sosednostna matrika cikla $C_n$. Potem velja:
    \begin{align*}
        A =  \begin{bmatrix} 
            0 & 1 & 0 & \dots & 1 \\ 
            1 & 0 & 1 & \dots & 0 \\ 
            0 & 1 & 0 & \dots & 0 \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            1 & 0 & 0 & \dots & 0 \end{bmatrix}
        =  \begin{bmatrix} 
            0 & 0 & 0 & \dots & 1 \\ 
            1 & 0 & 0 & \dots & 0 \\ 
            0 & 1 & 0 & \dots & 0 \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            0 & 0 & 0 & \dots & 0 \end{bmatrix} 
            +
            \begin{bmatrix} 
            0 & 1 & 0 & \dots & 0 \\ 
            0 & 0 & 1 & \dots & 0 \\ 
            0 & 0 & 0 & \dots & 0 \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            1 & 0 & 0 & \dots & 0 \end{bmatrix}
        = P + P^{-1}.
    \end{align*}
    Kjer je $P$ permutacijska matrika, ki predstavlja rotacijo vozlišč 
    cikla $C_n$ v smeri urinega kazalca.
    Nadaljno velja:
    \begin{align*}
        \det(P - \lambda I) &= -\lambda \begin{vmatrix} 
            -\lambda & 0 & 0 & \dots & 0 \\ 
            1 & -\lambda & 0 & \dots & 0 \\ 
            0 & 1 & -\lambda & \dots & 0 \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            0 & 0 & 0 & \dots & -\lambda  \end{vmatrix}
            + (-1)^{n+1} \begin{vmatrix} 
            1 & -\lambda & 0 & \dots & 0 \\ 
            0 & 1 & -\lambda & \dots & 0 \\ 
            0 & 0 & 1 & \dots & 0 \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            0 & 0 & 0 & \dots & 1  \end{vmatrix} \\
            &= (-1)^n \lambda^n + (-1)^{n+1}
    \end{align*}
    Posledično je karakteristični polinom matrike $P$ enak $\lambda^n - 1$, katerega ničle 
    so $n$-ti koreni enote. Lastne vrednosti matrike $A$ so zato enake 
    $\lambda_j = e^{i \frac{2\pi j}{n}} + e^{-i \frac{2\pi j}{n}} = 2\cosh(2\pi i j/n) = 2\cos(2\pi j/n)$.
\end{dokaz}


\begin{izrek}[Izrek o prepletanju lastnih vrednosti]\label{iz:interlacing}
Naj bo $G$ graf z $n$ vozlišči, ter $H$ njegov podgraf z $r$ vozlišči. Naj bo 
$A$ sosednostna matrika grafa $G$, ter $B$ sosednostna matrika grafa $H$. Naj bo
$\lambda_1 \geq \lambda_2 \geq \ldots \geq \lambda_n$ spekter matrike $A$, 
ter $\mu_1 \geq \mu_2 \geq \ldots \geq \mu_r$ spekter matrike $B$. Potem velja:
\[
\lambda_i \geq \mu_i \geq \lambda_{n - r + i} \quad \text{za vse } i \in \{1, 2, \ldots, r\}.
\]
\end{izrek}
\begin{dokaz}
    Dokaz izpustimo \cite{enwiki1325771674}.
\end{dokaz}

\begin{opomba}
Po lemi~\ref{le:cikellastne} zdaj vemo, da ima cikel $C_n$ natanko eno lastno vrednost enako $2$, 
in ker je $H[N_1(Q)]$ disjunktna unija ciklov (\ref{tr:unija_ciklov}), bo imel tak podgraf natanko toliko 
lastnih vrednosti enakih $2$, kolikor je v njem ciklov. % TODO: tukaj bi dal referenco na uvod kjer bi povedal da je spekter disjunktne unije ravno unija spetrov (očitno, ampak lepa referenca)

Izrek o prepletanju lastnih vrednosti~\ref{iz:interlacing} lahko nato uporabimo
direktno na podgrafu $H[N_1(Q)]$, vendar dobimo prešibke rezultate.

Izkaže se, da je bolje uporabiti izrek o prepletanju lastnih vrednosti na 
podgrafu $H[v \cup N_1(Q)]$, ki ga dobimo, če iz grafa $H$ odstranimo vsa vozlišča, 
razen enega iz $Q$ in tistih iz $N_1(Q)$. Nato pa opazovati spremembo večkratnosti
lastne vrednosti $2$ ko iz podgrafa $H[v \cup N_1(Q)]$ odstranimo vozlišče $v$.

V ta namen navedimo naslednjo definicijo in lemo, ki nam bosta služili pri nadaljnji analizi.
\end{opomba}

\begin{definicija}
    Naj bo $H$ poljuben graf z množico vozlišč $V(H)$, za
    poljubno $W \subset V(H)$ neprazno, definiramo delni spoj, označen z $\oplus$, 
    kot $v \oplus (H, W)$, kjer je $v$ novo vozlišče, ki je povezano z vsemi 
    vozlišči iz $W$, ter z nobenim drugim vozliščem grafa $H$.
\end{definicija}


\begin{lema}\label{le:knjiga}
    Naj bo $G = \cup_{i=1}^k C_{n_i}$, ter $W \subset V(G)$ neprazna. Potem je 
    multipliteta lastne vrednosti $2$ v grafu $v \oplus (G, W)$ enaka 
    $k-1$.
\end{lema}
\begin{dokaz}
    Vredno se je spomniti, da ima vsak cikel $C_n$ natanko eno lastno vrednost enako $2$ 
    (\ref{le:cikellastne}). Naj bodo $B_1, \ldots, B_k$ 
    sosednostne matrike ciklov $C_{n_1}, \ldots, C_{n_k}$
    ter $B$ sosednostna matrika grafa $G$.
    $B$ je bločno diagonalna matrika, katere bloki so 
    matrike $B_1, \ldots, B_k$. Posledično je multipliteta lastne vrednosti $2$ v 
    grafu $G$ enaka $k$.

    Naj bo dalje $A$ sosednostna matrika grafa $v \oplus (G, W)$.
    Podrobneje nas zanima dimenzija lastnega prostora za lastno vrednost $2$. 
    Označimo z $x_{C_{n_i}}$ vektor, ki ima na pozicijah, ki ustrezajo vozliščem cikla $C_{n_i}$, 
    vrednosti enake kot $\mathbf{x}$, in ničle drugod. Podobno označimo z $\mathbf{1}_{C_{n_i}}$ vektor, 
    ki ima enice zgolj na pozicijah, ki ustrezajo vozliščem cikla $C_{n_i}$. Potem velja:
    \begin{align*}
        A - 2I &= \begin{bmatrix} 
            B_1 - 2I & 0 & 0 & \dots & x_{C_{n_1}} \\ 
            0 & B_2 - 2I & 0 & \dots & x_{C_{n_2}} \\ 
            0 & 0 & B_3 - 2I & \dots & x_{C_{n_3}} \\ 
            \vdots & \vdots & \vdots & \ddots & \vdots \\ 
            x_{C_{n_1}}^\top & x_{C_{n_2}}^\top & x_{C_{n_3}}^\top & \dots & -2  \end{bmatrix}
            = \begin{bmatrix}
            B - 2I & \mathbf{x} \\
            \mathbf{x}^\top & -2
            \end{bmatrix}
        \end{align*}
    Naj bo zdaj $\begin{bmatrix}
        \mathbf{v} \\
        t
    \end{bmatrix} \in \ker(A - 2I)$, potem je $0 = (A - 2I) \begin{bmatrix}
        \mathbf{v} \\
        t
    \end{bmatrix} = \begin{bmatrix}
        (B - 2I)\mathbf{v} + t \mathbf{x} \\
        \mathbf{x}^\top \mathbf{v} - 2t
    \end{bmatrix}$. Ker $(B-2I)\mathbf{v} + t \mathbf{x} = 0$, je
    $-t\mathbf{x} \in \im(B-2I)$. Ker je $(B-2I)$ simetrična je $\im(B-2I) = \ker(B-2I)^\perp$,
    torej za vsak $u \in \ker(B-2I)$ velja $-t\mathbf{x} \cdot u = 0$. 
    Ker je $\ker(B-2I) = \{\mathbf{1}_{C_{n_1}}, \ldots, \mathbf{1}_{C_{n_k}}\}$ 
    (\ref{le:cikellastne}), in ker $W \cap V(G) \neq \emptyset$, 
    obstaja $i \in \{1, 2, \ldots, k\}$, da za $\mathbf{x} \cdot \mathbf{1}_{C_{n_i}} > 0$
    in posledično iz 
    $-t\mathbf{x} \cdot \mathbf{1}_{C_{n_i}} = 0$ sledi $t=0$.

    Sistem se poenostavi v $\begin{bmatrix}
        (B-2I)\mathbf{v} \\
        \mathbf{x}^\top \mathbf{v}
    \end{bmatrix} = 0$. Od tod velja $\mathbf{v} \in \ker(B-2I)$, torej 
    $\mathbf{v} = \alpha_1 \mathbf{1}_{C_{n_1}} + \ldots + \alpha_k \mathbf{1}_{C_{n_k}}$ in
    $\mathbf{x}^\top (\alpha_1 \mathbf{1}_{C_{n_1}} + \ldots + \alpha_k \mathbf{1}_{C_{n_k}}) = 0$.
    Ker je to ena enačba z $k$ spremenljivkami, je dimenzija rešitvenega prostora enaka $k-1$.
    Posledično je $\dim\ker(A-2I) = k-1$.
\end{dokaz}

Ker cikla $Q_i \cup P_j$ inducirata Petersenov graf,

z $\rho^0 = \id, \rho = (4\,0\,1\,2\,3), \rho^2 = (3\,4\,0\,1\,2), \ldots$

Cikle $Q_i$ in $P_j$ po potrebi preuredimo, zato
lahko predpostavimo $\pi_{0j} = \pi_{i0} = \id$.


Opazimo, da za $i, j \geq 1$ permutacija $\pi_{ij} \neq \id$.
Če bi bila, bi lahko našli cikel $C_4 = Q_i^k \overset{\pi_{ij}=\id}{\rightarrow}P_j^k \overset{\id}{\rightarrow} Q_0^k \overset{\id}{\rightarrow} P_0^k \overset{\id}{\rightarrow}Q_i^k$ (\ref{tr:mooreregularpaths}).

različne. Da se prepričamo BŠS vzemimo $\pi_{ij} = \pi_{i'j}$, kjer $\pi_{ij}(k) = k'$ --
našli bi kratek cikel
$C_4 = Q_i^k \overset{\pi_{ij}}{\rightarrow} P_j^{k'} \overset{\pi_{i'j}^{-1}}{\rightarrow} Q_{i'}^{k} \overset{\id}{\rightarrow} P_0^k \overset{\id}{\rightarrow}Q_i^k$ (\ref{tr:mooreregularpaths}).

vzemimo kar naravno izbiro $\pi_{i1} = \rho^i$ in $\pi_{1j} = \rho^j$. Opazimo, da je sedaj 
permutacija $\pi_{32}$ lahko bodisi $\rho$ ali $\rho^4$. Vendar pri izbiri $\pi_{32} = \rho^4$ 
je $\pi_{11}^{-1} \pi_{12} \pi_{32}^{-1} \pi_{31} = \rho^{-1} \rho^2 \rho^{-4} \rho^3 = \text{id}$, 
kar pomeni, da obstaja cikel $C_4 = P_1^k \overset{\rho^{-1}}{\rightarrow} Q_1^{k-1} \overset{\rho^2}{\rightarrow} P_2^{k+1} \overset{\rho^{-4}}{\rightarrow} Q_3^{k-3} \overset{\rho^3}{\rightarrow} P_1^k$ 
(\ref{tr:mooreregularpaths}). Od tu 
naprej lahko brez težav zapolnimo preostale permutacije, saj so si vse permutacije v istem stolpcu ali vrstici 
različne (\ref{fig:hoffman_permutacije}).

ki ga imenujemo Hoffman-Singeltonov graf.
Preostane nam le še pokazati, da je graf resnično Mooreov in v zadnjih korakih nismo ustvarili
kratkih ciklov.
\end{opomba}

\begin{opomba}\label{op:sosedstva_ciklov}
    Permutacije $\pi_{ij} = \rho^{ij \pmod 5}$, ki smo jih našli, nam povedo, da med cikli povezave potekajo po naslednjem pravilu.
        \begin{align*}
        \forall i, j, k, m:\; &Q^u_i \sim P_j^{u + i\cdot j \pmod 5} \\
        &P^v_j \sim Q_i^{v - i\cdot k \pmod 5}
    \end{align*}
\end{opomba}

Videli smo, da $Q_i \cup P_j$ inducira Petersenov graf, ki 
je seveda Mooreov (\ref{sec:petersenov_graf}) z ožino $5$. Torej, če bi kratek
cikel obstajal, bi moral biti potekati vsaj skozi enega $P_i$-ja in dva $Q_i$-ja (ali obratno).
Ampak če cikel obišče natanko enega $P_i$-ja in dva $Q_i$-ja, potem 
imamo najmanj $P_i^k \overset{\pi_{ij}}{\rightarrow} Q_j^{k'} Q_j^{k'\pm1} \overset{\pi_{ij}^{-1}}{\rightarrow} P_i^{k''\neq k} \overset{\pi_{ij'}}{\rightarrow} Q_{j'}^{k'''} Q_{j'}^{k'''\pm 1} \overset{\pi_{ij'}^{-1}}{\rightarrow} P_i^k$.
Cikla ne moremo zaključiti pred $6$-imi koraki.

Edina ostala možnost, ki bi lahko ustvarila cikel, je, da obišče natanko dva $P_i$-ja in dva $Q_i$-ja,
in s tem tvori $C_4$. Recimo, da je ta cikel oblike $Q_i^u P_j^v Q_k^w P_l^x Q_i^u$.

Skličemo se na opombo \ref{op:sosedstva_ciklov}, ki pove, da iz vozlišča $Q_i^u$ lahko pridemo v $P_j^v$ 
le, če je $v = u + ij \pmod 5$. Podobno se nazaj vračamo z inverzno permutacijo, in zato
iz vozlišča $P_j^v$ pridemo v $Q_k^w$ zgolj, če je $w = v - jk \pmod 5$.

Naš cikel $Q_i^u P_j^v Q_k^w P_l^x Q_i^u$ bi tako lahko obstajal le, če bi veljal sistem naslednjih enačb:
\begin{align}
    v &\equiv u + ij \pmod 5 \\
    w &\equiv v - jk \pmod 5 \\
    x &\equiv w + kl \pmod 5 \\
    u &\equiv x - li \pmod 5
\end{align}
Vendar seštete enačbe dajo $v + w + x + u \equiv v + w + x + u + ij - jk + kl - li \pmod 5$, 
kar se poenostavi v $0 \equiv (i-k)(j-l) \pmod 5$. Potem je bodisi $(i-k) \equiv 0$
bodisi $(j-l) \equiv 0$. Ker so $i, j, k, l$ elementi množice $\{0, 1, 2, 3, 4\}$,  
to prisili, 
da je $i=k$ ali $j=l$, kar pomeni, da tudi tak cikel ne obstaja.

Tako smo pokazali, da je graf, ki smo ga konstruirali, ne le resnično Mooreov graf stopnje $7$ in 
premera $2$, temveč tudi unikaten.


\paragraph{Grupa avtomorfizmov}
\begin{trditev}\label{tr:red_avtomorfizmov}
    Red avtomorfizmov Hoffman-Singeltonovega grafa je $|\aut(H)| = 50\cdot 7!$.
\end{trditev}%aaaaa

Fiksirali bomo nek cikel $C_5$ v grafu $H$, recimo $P_0$ in opazovali njegovo orbito (\ref{def:orbita})
ter stabilizatorsko grupo (\ref{def:stabilizator}). Če označimo $G = \aut(H)$, nam izrek o 
orbitah in stabilizatorjih~\ref{op:orbita_stabilizator_red}
pove, da je $|\aut(H)| = |G\cdot P_0||G_{P_0}|$.

Število elementov v orbiti $G\cdot P_0$ je očitno enako številu ciklov $C_5$ v grafu $H$,
v katere lahko preslikamo $P_0$.
Red stabilizatorja $G_{P_0}$ pa bomo izračunali tako, da bomo našli grupo $\Pi$, ki
je v njem vsebovana in pokazali, da mu je v resnici enaka.

\begin{lema}
    Za red orbite cikla $P_0$ velja $|G\cdot P_0| = 50\cdot 7\cdot 6\cdot 6$. Število ciklov $C_5$ v 
    grafu $H$ je enako $|G\cdot P_0|/10 = 1260$.
\end{lema}
\begin{dokaz}
    Ker ima graf $H$ premer $2$ in ožino $5$ (\ref{tr:mooreregularpaths}), 
    vsaka pot dolžine $4$ enolično določa
    cikel $C_5$. Tako pot lahko začnemo v poljubnem vozlišču (enem izmed $50$-ih). Nadaljujemo
    jo s poljubnim sosedom (enem izmed $7$-ih). Nato pa z vozliščem, ki ni sosed prvega, 
    ampak je sosed drugega (enem izmed $6$-ih), itd.
    Red orbite je tako določen, vendar cikle smo tu šteli vred z vsemi njihovimi rotacijami, ki jih je $10$.
\end{dokaz}

\begin{lema}\label{le:permutira_qi_je}
    Vsak avtomorfizem, ki fiksira $P_0$ (\ref{def:fiksni_element}) permutira $Q_i$-je, brez rotacij.
\end{lema}
\begin{dokaz}
    Naj bo $\sigma \in G_{P_0}$ in $Q = \{Q_0, \ldots, Q_4\}$. 
    Potem je $P_0^j \sim Q_k^j$ (\ref{op:sosedstva_ciklov}).
    Tako za presekano množico sosedov velja $N(P_0^j)\cap Q = \{Q_k^j \mid k \in \mathbb{Z}_5\}$.
    Ker $\sigma$ ohranja sosedstva in fiksira vsa vozlišča $P_0^j$ velja 
    $\sigma(\{Q_k^j \mid k \in \mathbb{Z}_5\}) = \{Q_k^j \mid k \in \mathbb{Z}_5\}$
    (\ref{tr:stabilizator_sosedstvo}), kar pomeni, da $\sigma$ kvečjem permutira $Q_i$-je brez rotacij.
\end{dokaz}

Zdaj vemo, da lahko vsak avtomorfizem, ki fiksira $P_0$, preslika $Q_k^j$ zgolj v $Q_{k'}^j$ za nek $k'$.
Ker pa mora ohraniti tudi povezanost, mora vsak $P_i^j$ preslikati v nak natanko določen $P_{i'}^{j'}$.
Zdaj lahko izberemo dve taki neodvisni presilkavi $\alpha$ in $\beta$, 
ki fiksirata $P_0$, in z njima generiramo grupo
$\Pi = \langle\alpha,\beta\rangle$. Izberemo
$\alpha, \beta: V(H) \to V(H)$, definirani kot:
    \begin{align}
        &\alpha: \begin{cases}
            P_j^i &\mapsto P_j^{3i} \\
            Q_j^k &\mapsto Q_j^{2k}
        \end{cases}\\
        &\beta: \begin{cases}
            P_j^i &\mapsto P_{j-i}^i \\
            Q_j^k &\mapsto Q_j^{k+1}
        \end{cases}
    \end{align}

\begin{opomba}
    Preslikavi $\alpha$ in $\beta$ seveda ohranjata sosedstva ter fiksirata vozlišča cikla $P^0$,
    zato sta avtomorfizma grafa $H$ in vsebovana v stabilizatorski grupi $G_{P^0}$.
    Posledično je tudi grupa, ki jo generirata, $\Pi \leq G_{P^0}$. Preostane nam le še pokazati,
    da je tudi $G_{P^0} \leq \Pi$, torej da je $\Pi = G_{P^0}$.
\end{opomba}

\begin{lema}\label{le:red_grupe_pi}
    Grupa $\Pi$ je izomorfna semidirektnemu produktu 
    (\ref{def:semidirektni_produkt}) cikličnih grup (\ref{def:ciklicna_grupa}) $C_5$ in $C_4$.
    Posledično je $|\Pi| = 20$ (\ref{tr:semidirect_order}).
\end{lema}
\begin{dokaz}
    Najprej preverimo, da sta $\alpha$ in $\beta$ resnično avtomorfizma grafa $H$. 
    Po opombi~\ref{op:sosedstva_ciklov} je $P^i_j \sim Q^k_{j + ik}$.
    Res preslikavi ohranjata sosedstva, saj 
    \begin{align*}
    &\alpha(P^i_j) = P^{3i}_j \sim Q^{2i}_{j + (3i)(2k)} 
        \equiv Q^{2i}_{j + ik} = \alpha(Q^k_{j + ik}) \\
    &\beta(P^i_j) = P^i_{j-i} \sim Q^{k+1}_{(j-i)+i(k+1)} 
        \equiv Q^{k+1}_{j + ik} = \beta(Q^k_{j + ik}).
    \end{align*}

    Dalje pokažimo, da je $\beta$ normalna podgrupa (\ref{def:normalna_podgrupa}) v $\Pi$.
    Velja namreč
    \begin{align*}
    &(\alpha\circ\beta\circ\alpha^{-1})(V(H)) \ni (\alpha\circ\beta\circ\alpha^{-1})(P^i_j) = (\alpha\circ\beta)(P^{2i}_j) = 
    \alpha(P^{2i}_{j-2i}) = P^{3(2i)}_{j-2i} \\
    &\equiv P^i_{j-2i} = \beta^2(P^i_j) \in \langle\beta\rangle \\
    &(\alpha\circ\beta\circ\alpha^{-1})(V(H)) \ni (\alpha\circ\beta\circ\alpha^{-1})(Q^k_j) = (\alpha\circ\beta)(P^{3k}_j) = 
    \alpha(P^{3k}_{j-3k}) = P^{3(3k)}_{j-3k} \\
    &\equiv P^k_{j-3k} = \beta^3(Q^k_j) \in \langle\beta\rangle
    \end{align*}

    Ker je tudi $\langle\beta\rangle \cap \langle\alpha\rangle = \{\id\}$,
    namreč $\alpha$ fiksira $Q^0$ medtem ko ga $\beta$ nikoli ne,
    sedaj vemo, da je $\Pi = \langle\beta\rangle \rtimes \langle\alpha\rangle$, 
    semidirektni produkt (\ref{def:semidirektni_produkt}), in da
    velja $|\Pi| = |\langle\beta\rangle||\langle\alpha\rangle|$ 
    (\ref{op:lastnost_semidirektnega_produkta}).

    Ker je $\beta^5 = \id$ in $\alpha^4 = \id$, je $|\langle\beta\rangle| = 5$ in 
    $|\langle\alpha\rangle| = 4$, posledično je $|\Pi| = 20$.

    Mimogrede, ker je $\alpha$ edini generator grupe $\langle\alpha\rangle$ in je reda
    $4$, je $\langle\alpha\rangle$ izomorfna ciklični grupi $C_4$. 
    Podobno je $\langle\beta\rangle$ izomorna ciklični grupi $C_5$. Tako tudi 
    $\Pi \cong C_5 \rtimes C_4$.
\end{dokaz}


\begin{lema}
    Grupa $\Pi$ je natanko stabilizatorska grupa $G_{P^0}$.
    Posledično je $|G_{P^0}| = 20$.
\end{lema}
\begin{dokaz}
    Če bi $G_{P^0}$ imel še kako permutacijo $\tau \not\in \Pi$,
    potem bi ta prisilila obstoj netrivialne permutacije, ki bi fiksirala $P^0$
    in še dva $Q^i$-ja. Namreč po Lagrangevem izreku~\ref{iz:lagrangev_izrek} 
    vemo, da je $|G_{P^0}|$ večkratnik reda svoje podgrupe $\Pi$, torej večkratnik
    števila $20$ (\ref{le:red_grupe_pi}). Če je grupa $G_{P^0}$ večja od grupe $\Pi$, 
    mora veljati $|G_{P^0}| \geq 2\cdot20 = 40$. Po izreku o orbitah in 
    stabilizatorjih~\ref{op:orbita_stabilizator_red} 
    je $|G_{P^0}| = |G_{P^0}\cdot Q^0||(G_{P^0})_{Q^0}|$.
    Ker $G_{P^0}$ fiksira $P^0$, permutira zgolj $Q^i$-je brez 
    rotacij (\ref{le:permutira_qi_je}), se $Q^0$ pri fiksiranem
    $P^0$ lahko preslika natanko v enega izmed $5$-ih $Q^i$-jev, 
    zato $|G_{P^0}\cdot Q^0| = 5$.

    Velja torej $|(G_{P^0})_{Q^0}| \geq 40 / 5 = 8$. Zdaj izrek o orbitah 
    in stabilizatorjih~\ref{op:orbita_stabilizator_red} uporabimo še enkrat, 
    in tokrat dobimo 
    $|(G_{P^0})_{Q^0}| = |(G_{P^0})_{Q^0}\cdot Q_1||((G_{P^0})_{Q^0})_{Q_1}|$
    in posledično $|((G_{P^0})_{Q^0})_{Q^1}| \geq 8 / 4 = 2$.
    To nam res neposredno pove, da obstaja vsaj ena netrivialna permutacija 
    $\sigma \in G_{P^0}$, ki fiksira $P^0$, $Q^0$ in $Q^1$. Označimo torej lahko
    $S = \{P^0, Q^0, Q^1\}$ in $\sigma(S) = S$. 

    Preostane nam še pokazati, da bi taka permutacija $\sigma$ nujno ustvarila
    cikel $C_4$, kar bi bilo protislovje.

    Vzemimo poljubno vozlišče $P^i_j$, kjer $i\neq 0$. Potem sta njegova soseda v množici 
    $S$ enaka $N(P^i_j)\cap S = \{Q^0_j, Q^1_{j+1}\}$ (\ref{op:sosedstva_ciklov}). 
    Ker $\sigma$ fiksira $S$ 
    točkovno, velja $N(\sigma(P^i_j)) \cap S = N(P^i_j) \cap S$, %% TODO: Utemelji. z referenco.
    torej si $\sigma(P^i_j)$ in $P^i_j$ delita soseda $Q^0_j$ in $Q^1_{j+1}$.
    Tako smo prispeli v protislovje, našli smo namreč cikel
    $C_4 = P^i_j Q^0_j \sigma(P^i_j) Q^1_{j+1} P^i_j$. 
\end{dokaz}

\begin{dokaz}[Dokaz trditve~\ref{tr:red_avtomorfizmov}]
Po razmisleku iz uvoda sekcije skupaj s prejšnjimi ugotovitvami lahko res zaključimo, da velja
 $|\aut(H)| = |G\cdot P^0||G_{P^0}| = (50\cdot 7\cdot 6\cdot 6) \cdot 20 = 50 \cdot 7!$.
\end{dokaz}

\begin{definicija}\label{def:rang_grupe}
    Pravimo, da ima grupa $G$ na množici $\Omega$ rang $r$, če je $r$ število orbit, ki jih 
    ima stabilizator $G_v$  (\ref{def:stabilizator}) poljubne točke $v \in \Omega$ 
    (ekvivalentno $r = |\Omega / G_v|$). Drugače povedano,
    če ekvivalenčna relacija (\ref{def:ekvivalencna_relacija}) $y\sim z \iff y \in G_v \cdot z$ 
    povzroči razpad na $r$ ekvivalenčnih razredov (\ref{def:ekvivalencni_razredi}).

    Rang grupe je dobro definiran zgolj v primeru, ko je neodvisen od izbire vozlišča $v$, torej ko 
    grupa $G$ deluje tranzitivno na množici $\Omega$ (\ref{def:tranzitivno_delovanje}).
\end{definicija}

\subsection{Neobstoj Mooreovih grafov}

\citeauthor{hoffman1960moore} \cite{hoffman1960moore} pokažeta, da Mooreovi grafi za premer $3$ ne obstajajo,
z izjemo trivialnega primera $C_7$. 

\paragraph{Štetje sprehodov}
\citeauthor{plesnik1974one} \cite{plesnik1974one} razvije metodo s katerimi z računalniško asistenco
pokaže, da Mooreovi grafi za premer $4$ ne obstajajo vse do neke meje.

$F_n$ označuje število sprehodov dolžine $n$, kjer prva in zadnja povezava poteka po zadnjem nivoju $N_D(w)$.
Očitno mora tako po ostalih nivojih potekati sodo število (recimo $2t$) povezav. Prvo povezavo (po $N_D(w)$) lahko 
izberemo na $(d-1)$ načinov, potem lahko nadaljujemo po ostalih nivojih. Ko se nižamo proti $w$ nimamo 
se lahko nižamo le po eni poti, ter ko se vračamo proti $N_D(w)$ imamo $(d-1)$ možnosti. Tako s povezavami
pri teh ostalih nivojih doprinesemo še faktor $(d-1)^t$. Ker nam je zdaj preostalo še $n-2t-1$ korakov
preden zaključimo sprehod, v množici $N_D(w)$ dobimo še faktor $(d-2)^{n-2t-1}$, saj ne smemo izbrati poti, 
po kateri bi se vrnili v začetno vozlišče. Po takšnih premislekih je Plesnik dobil naslednjo identiteto.
\begin{align}
&N(t, n) = (\Delta-1)^{t+1}(\Delta-2)^{n - 2t -1} \\
&F_n = \sum_{t=0}^{\lfloor (n-2)/2 \rfloor} \binom{n-t-2}{t} N(t, n)\quad \forall n \geq 2
\end{align}

Potem izračuna, za vsak rob grafa, število ciklov dolžine $2D + n$, ki ga označi z $r_{2D+n}$, in dobi:
\begin{equation}
    r_{2D+n} = r_{2D + n-2} + (\Delta-1)^{D-1}(F_n - F_{n-2}) \quad \forall 1 \leq n \leq 2D+1
\end{equation}

Potem dokaže še preprosto dejstvo, da če je nek rob vsebovan v natanko $r$ podgrafih, ki so vsi cikli dolžine $2D+1$,
in je teh podgrafov $s$ in je v vsakem $m$ povezav,
potem velja $r |E| = sm$ oziroma $r |E| \equiv 0 \pmod m$.

Po tem malem dokazu, očitno sledi njegov izrek, ki pravi
\begin{equation}
    \frac{1}{2} (1 + \Delta\sum_{i=0}^{D-1} (\Delta-1)^i) r_{2D+n} \equiv 0 \pmod {2D+n} 
    \quad\forall 1 \leq n \leq 2D+1
\end{equation}
S pomočjo te enačbe, ki jo z Fermatovim izrekom še oplemeniti, računalniško ovrže obstoj Mooreovih 
grafov za parametra $(D, \Delta) \in [3, 100]^2 \cup (\{3\} \times [3, 1000])$.


\paragraph{Krotenje lastnih vrednosti}
\citeauthor{bannai1973finite} \cite{bannai1973finite} zaključita vprašanje o neobstoju, in pokažeta,
da z izjemo trivialnih primerov, Mooreovi grafi za premer $D \geq 3$ ne obstajajo.

Če močno povzamemo, to storita tako, da uporabita dejstvo, da so Mooreovi grafi razdaljno regularni,
torej opazujeta njegovo presečno matriko
\begin{equation*}
    M = \begin{bmatrix}
0 & D & 0 & 0 & \cdots & 0 \\
1 & 0 & D-1 & 0 & \cdots & 0 \\
0 & 1 & 0 & D-1 & \cdots & 0 \\
0 & 0 & 1 & 0 & \ddots & \vdots \\
\vdots & \vdots & & \ddots & \ddots & D-1 \\
0 & 0 & \cdots & 0 & 1 & 0
\end{bmatrix}
\end{equation*}
Vesta, da se njen karakteristični polinom ujema z minimalnim polinomom matrike sosednosti Mooreovega grafa.
Njen karakteristični polinom lahko dobimo z izračunom $G_D := \det(xI - M)$. Dobimo rekurzijo
$f_D() = xf^{D-1} - (D-1)f^{D-2}$, z začetnimi pogoji $f^0 = 1$ in $f^1 = x$. Izkaže se, da obstaja 
povezava med 
$G^{\Delta, D}(x) = f^D(x) + (D-1)f^{D-1}(x)$.

Od tu dalje uporabita formulo, ki jo poda Higman \cite[lema 3.4]{feit1964nonexistence}, 
ki pravi, da $m(\theta_i) = \frac{\sled(f^D_i(A))}{f^D_i(\theta_i)}$, 
kjer je $f^D_i(x) = f^D(x)/(x-\theta_i)$.

Izračunata, da je 
$m(\theta_i) = \frac{Df^{D-1}(\theta_i)}{(\theta_i-D)\frac{\partial G^{\Delta, D-1}}{\partial x}(\theta_i)}$.
Tako sta z substitucijo $x = 2s\cos(\phi)$ in $s^2 = D - 1$ izrazila $m(\theta_i)$ kot trigonometrično funkcijo.

Z analizo ničel funkcije $G^{\Delta, D}(x)$ (ki se ujemajo z lastnimi vrednostmi Mooreovega grafa)
pokažeta, da so lastne vrednosti razporejene dokaj enakomerno in sta našla intervalske omejitve zanjih.

Potem jima je preko formule za $m(\theta_i)$ preko razmerja večkratnosti po mnogo analitičnih neenakostih 
uspelo pokazati, da sta večkratnosti od največje lastne vrednosi (izklučajoč trivialno lastno vrednost $D$)
$\theta_1$ in najmanjše lastne vrednosti $\theta_D$ manjše od vseh ostalih. 

Potem je sledilo, da morata biti algebrajsko konjugirani. Pokazala sta, da sta bodisi obe iracionalni
in potem bi moralo biti $\theta_1 + \theta_D \in \mathbb{Z}$ kar je bilo protislovje, ali pa 
da sta obe racionalni, vendar sta že za $\theta_D$ pokazala, da je interval okoli nje tako majhen, 
stran od celih števil, da ne dopušča celosti, razen v nekih omejenih primerih, ki sta jih preverila 
računalniško.

Za Mooreov graf velja, da ne le doseže maksimalno število vozlišč za dano maksimalno
stopnjo $\Delta$ ter premer $D$, temveč tudi doseže minimalno število vozlišč za dano
minimalno stopnjo $\Delta$ ter ožino $g=2D + 1$. Spodnjo mejo za število vozlišč za graf
z dano minimalno stopnjo $\Delta$ in ožino $g$ lahko naravno razširimo tudi za sode ožine.

Začnimo z dvema povezanima vozliščema $u$ in $v$, ko začnemo potovati iz vozlišča $u$ v eno smer
ter z vozlišča $v$ v drugo, se ti poti ne smeta srečati pred $k-1$ koraki -- namreč to bi vodilo 
v protislovje z izbrano ožino $g=2k$, ker pa minimiziramo število povezav predpostavimo, da se 
poti stakneta na $k$-tem koraku. Tako smo začeli z ničelnim nivojem, kjer smo imeli dve vozlišči $u$ 
in $v$, v prvem nivoju smo tako imeli $2(k - 1)$ novih vozlišč, saj je minimalna stopnja enaka $k$,
ter zaradi minimiziranja števila vozlišč je graf $k$ regularen. V $i$-tem nivoju imamo $2(k-1)^i$ 
vozlišč. Tako smo izpeljali spodnjo mejo enako 
$2\sum_{i=0}^{k-1} (k - 1)^i = 2\frac{(\Delta-1)^k - 1}{k - 2}$.

\begin{trditev}\label{tr:transition}
Naj bo $G$ regularen graf stopnje $\Delta$, ter z dolžino 
najkrajšega cikla, oziroma ožino, $g$. Tedaj obstaja spodnja meja 
za red grafa $G$ ter je enaka:
\begin{equation} %V opisu diplomske naloge je bila napaka.%
    |V(G)| \geq \begin{cases}
    \frac{\Delta(\Delta - 1)^{(g-1)/2} - 2}{\Delta - 2}, & g \text{ je liha} \\
    \frac{2(\Delta - 1)^{g/2} - 2}{\Delta - 2}, & g \text{ je soda}
    \end{cases}
\end{equation}
\end{trditev}

V nadaljevanju vzemimo trditev~\ref{tr:transition} kot razširitev Moorevih grafov, pravimo 
da so tudi grafi, ki za dano ožino $g$ ter minimalno stopnjo $\Delta$ dosežejo minimalno število 
vozlišč, Mooreovi oziroma razširjeno Mooreovi.

\subsection{Geometrije}
\begin{definicija}
Geometrija (ranga n) je nabor
$\Gamma = \left(
    \A_1, \A_2, \ldots, \A_n, \II
\right)$, kjer so $\A_1, \A_2, \ldots, \A_n$ neprazne paroma disjunktne 
množice, in $\II$ relacija incidenčnosti za katero velja 
\[\II \subseteq \bigcup_{i\neq j} \A_i \times \A_j.\] Če za poljubna
$a, b \in \bigcup_{i} \A_i$ velja $(a, b) \in \II$, potem pišemo tudi
$\IIw{a}{b}$. 

Če za neko pozitivno celo število $m$ velja še 
\[\forall i \; \forall a \in \A_i\; \forall j \neq i : \ 
m = \#\{b \in \A_j\mid \IIw{a}{b} \},\]
potem pravimo, da je geometrija reda $m$.
\end{definicija}

\begin{definicija}
Naj bo $\Gamma = \left(
    \A_1, \A_2, \ldots, \A_n, \II
\right)$ geometrija. Pravimo, da je $\Gamma^D$ njena dualna geometrija 
če je $\Gamma^D = \left(
    \A_1, \A_2, \ldots, \A_n, \II^D
\right)$, kjer je $\IIwd{a}{b} \Leftrightarrow \IIw{b}{a}$.
\newline 
Podobno pravimo, da je $\Gamma'$ podgeometrija natanko 
tedaj ko $\Gamma' = \left(\A_1', \A_2', \ldots, \A_n', \II'\right)$,
kjer za vsak $1 \leq i \leq n$ velja $\A_i' \subseteq \A_i$ ter kjer 
je $\II'$ ustrezna zožitev relacije $\II$.
\end{definicija}



Navkljub splošnejšim definicijam si podrobneje poglejmo geometrije ranga 2
in strukture, ki le tam nastanejo.

\begin{definicija}
Geometrija (ranga 2) je nabor 
$\Gamma = \left(\PP, \LL, \II\right)$. Množici $\PP$ pravimo \textsl{točke}, 
množico $\LL$ pa poimenujmo \textsl{črte}. Pravimo, da je $p \in \PP$ 
\textsl{točka} ter $l \in \LL$ \textsl{črta}. \newline 
\textsl{Črta} je v relaciji z \textsl{točko}, če velja $\IIw{l}{p}$.
\end{definicija}

\begin{definicija}[Incidenčni graf]
Naj bo $\Gamma = \left(
    \A_1, \A_2, \ldots, \A_n, \II

$E(G) = \{uv \mid \IIw{u}{v} \lor \IIw{v}{u}\}$. Tak graf je očitno $n$-delen.
\end{definicija}

\subsubsection{Projektivne ravnine}

\begin{definicija}[Projektivna ravnina]
    Projektivna ravnina je incidenčna struktura drugega ranga $(\PP, \LL, \II)$ 
    sestavljena iz množice točk $\PP$, množice premic $\LL$ ter incidenčne relacije
    $\II \subset \PP \cup \LL$, ki zadostuje naslednjim lastnostim:
    \begin{itemize}
        \item Za dve različni točki obstaja natanko ena premica, ki je incidenčna z obema.
        \item Za dve različni premici obstaja natnako ena točka, ki je z njima incidenčna.
        \item Obstajajo štiri take točke, da nobena premica ni incidenčna z več kot dvema izmed njih.
    \end{itemize}
\end{definicija}

Ni se težko prepričati, da ta definicija povsem sovpada z definicijo posplošenih trikotnikov, 
če le tej niso reda $(1, 1)$.

\paragraph{Desarguesove projektivne ravnine}
\begin{definicija}\label{def:desarguesova_ravnina_verzija_1}
Desarguesova projektivna ravnina je projektivna ravnina, ki zadostuje Desarguesovemu aksiomu.
Moč jo je konstruirati iz obsega z deljenjem (ref...) $\mathbb{K}$, tako da definiramo 
točke $\PP$ kot trojice $(a, b, c) \in \mathbb{K}^3 \setminus \{(0,0,0)\}$, 
kjer so vse točke, ki se razlikujejo zgolj za večkratnik smatrane kot enake;

premice $\LL$ kot trojice $[a, b, c] \in \mathbb{K}^3 \setminus \{(0,0,0)\}$, 
kjer so prav tako tiste, ki se razlikujejo le za večkratnik smatrane kot enake;

ter incidenčno relacijo, ki pravi, da je točka $(x, y, z)$ incidenčna s premico $[u, v, w]$
natanko tedaj, ko $ux + vy + wz = 0$.
\end{definicija}

\begin{definicija}\label{def:desarguesova_ravnina_verzija_2}
Alternativno si jih lahko predstavljamo kot geometrijo drugega ranga $(\PP, \LL,\II)$, 
kjer dodamo točke v neskončnosti ($(\infty)$ in $(m)$), 
ter premice v neskončnosti ($[\infty]$ in $[m, k]$), tako, da se poljubni dve premici 
(četudi vzporedni)
sekata v natanko eni točki.

Tako se vse navpične premice $[x]$ sekajo v točki v neskončnosti $(\infty)$ in točkah $(x, y)$,
vse smerne premice $[m, k]$ sekajo v točki v neskončnosti $(m)$ in točkah $(x, y)$ natanko
tedaj ko je $y = mx + k$.
Na premici v neskončnosti $[\infty]$ pa ležijo vse točke v neskončnosti $(\infty)$ in $(m)$.
\end{definicija}

\begin{trditev}
    Vse končne Desarguesove projektivne ravnine so 
    konstruirane nad polji ($\mathbb{K} = \mathbb{F}_{p^n}$).
\end{trditev}
\begin{dokaz}
    Wedderburnov izrek (1905) pravi, da je vsak končni obseg (devision ring) je končno 
    polje $\mathbb{F}_q$ za neko $q = p^n$, kjer je $p$ praštevilo in $n$ naravno število.
    Po definiciji Desarguesove projektivne ravnine 
    (\ref{def:desarguesova_ravnina_verzija_1}) lahko vsako Desarguesovo projektivno
    ravnino konstruiramo nad nekim obsegom, ki pa je po Wedderburnovem izreku končno polje.
\end{dokaz}

\paragraph{Abstraktne projektivne ravnine}
\begin{definicija}\label{def:planarni_ternarni_obroc}
    Planarni ternarni obroč $(R, T)$ je algebraična struktura (\ref{def:algebraicna_struktura}),
    kjer je $R$ množica vsaj dveh elementov, imenovanih $0$ in $1$, ter 
    $T: R^3 \to R$ ternarna operacija, ki zadostuje naslednjim lastnostim:
    \begin{enumerate}[label=(\roman*)]
        \item $T(a, 0, b) = T(0, a, b) = b \quad \forall a, b \in R$,
        \item $T(1, a, 0) = T(a, 1, 0) = a \quad \forall a \in R$,
        \item $\forall a, b, c, d \in R,\, a \neq c \implies \exists! x \in R:\; T(x, a, b) = T(x, c, d)$,
        \item $\forall a, b, c \in R,\; \exists! x\in R:\; T(a, b, x) = c$,
        \item $\forall a, b, c, d\in R, a\neq c, \;\exists! (x, y) \in R^2:\; T(a, x, y) = b\,\land\,T(c, x, y) = d$
    \end{enumerate}

    Velja celo to, da ne obstaja noben drug par $(0', 1') \in R^2$ za katerega bi $T$ zadoščala aksiomom,
    in da v primeru končne množice $R$ velja $(\textit{iv})\implies ((\textit{iii})\iff(\textit{v}))$.
    \end{definicija}
\begin{zgled}
    Pri definiciji ternarnega obroča (\ref{def:planarni_ternarni_obroc}), si lahko mislimo, 
    da enačba $T(a, x, y) = b$ pravi, da točka $(a, b)$ leži na premici
    z nagibom $x$ in odsekom $y$ označeni (oziroma na premici $[x, y]$).
    
    Potem lahko zahteve razumemo na naslednji način: 
    \begin{enumerate}[label=(\roman*)]
        \item zagotavlja, da na vodoravni premici $[0, b]$ leži vsaka točka tipa $(a,b)$
    (predstavljajmo si $b = 0\cdot a + b$) in hkrati, da vsaka običajna
    premica $[a, b]$ seka navpično os v točki $(0, b)$ (predstavljajmo si $b = a\cdot 0 + b$).
    \item določi merilo, zagotavlja, da na 
    glavni diagonali $[1, 0]$ leži vsaka točka tipa $(a,a)$ (predstavljajmo si $a = 1\cdot a + 0$),
    medtem ko na premici z nagibom $a$ skozi izhodišče ($[a,0]$) leži točka $(a, 0)$
    (predstavljajmo si $0 = a\cdot 0 + 0$).
    \item zagotavlja, da če izberemo nagib $a$ in točko $(b,c)$, obstaja natanko ena premica
    $[a, x]$ s tem nagibom, na kateri leži izbrana točka (predstavljamo si $c = ab + x$).
    \item zagotavlja, da se poljubni premici z različnima nagiboma ($a\neq c$) sekata
    v natanko eni točki $(x,y)$ (predstavljajmo si $y = ax + b = cx + d$).
    \item zagotavlja, da za poljubni dve točki $(a, b)$ in $(c, d)$ z različnima 
    $x$-koordinatama ($a\neq c$) obstaja natanko ena premica $[x,y]$ 
    na kateri ležita obe hkrati.
    \item 
    \end{enumerate}

    Tako vidimo, da $T(x, m, k) = mx + k$ zadošča vsem aksiomom. Taka funkcija
    pa se pojavi tudi pri definiciji Desarguesove
    projektivne ravnine s pomočjo točk v neskončnosti 
    (\ref{def:desarguesova_ravnina_verzija_2}). 
    
    Opazimo lahko, da če v definiciji Desarguesove projektivne ravnine 
    (\ref{def:desarguesova_ravnina_verzija_2}) nadomestimo $y = mx + k$ z $y = T(x, m, k)$,

\end{zgled}

\begin{opomba}

\end{opomba}

\begin{itemize}

    \item ${((a, b), [c])\in\II\iff a = c}$
    \item ${((a, b), [\infty])\not\in\II}$
    \item ${((a), [c, d])\in\II\iff a = c}$
    \item ${((a), [c])\not\in\II}$

\begin{izrek}[Bruck-Ryserjev izrek]
    Če je projektivna ravnina končna, obstaja število $N$, da vsaka premica vsebuje $N+1$ točk, 
    in je vsaka točka vsebovana v $N+1$ premicah. Če je $N \equiv 1$ ali $2 \mod 4$, ter če
    brezkvadratni del števila $N$ vsebuje vsaj en prafaktor oblike $4k + 3$, potem ne obstaja
    končna projektivna ravnina z $N+1$ točkami na premici \cite{bruck1949nonexistence}. Po 
    Fermatovem izreku o vsoti dveh kvadratov \cite{enwiki1319561489} je pogoj s prafaktorjem ekvivalenten temu,
    da je brezkvadratni del vsota dveh kvadratov.
\end{izrek}

\begin{definicija}[Galoisovo polje]
    Galoisovo polje je množica s končnim številom elementov $s$, v kateri veljajo operacije
    seštevanja, odštevanja, množenja in deljenja (razen z ničelnim elementom). 
    Označeno ga z $\GF(s)$. Obstaja natanko takrat ko je $s=p^n$ za poljubno praštevilo $p$ in 
    naravno število $n$. 
\end{definicija}

\begin{opomba}
    Če je $p$ praštevilo je $\GF(p) = \mathbb{Z}_p := \{0, 1, 2, \dots, p - 1\}$. 
    Dodatno je $\GF(p^n) = \{a_{n-1}x^{n-1} + \dots + a_1x + a_0 \mid a_i \in \GF(p)\}$ 
    množica polinomov, kjer množenje definiramo tako, da računamo v modulu nerazcepnega polinoma
    $P(x)$ stopnje $n$ (torej polinom, ki se ga v $\GF(p^n)$ ne da razbiti na produkt drugih).
    Tako velja $\GF(p^n) \cong \GF(p)[x] \,/\, \langle P(x) \rangle$.
\end{opomba}

\subsubsection{Posplošeni štirikotniki}

\begin{definicija}\label{def:posploseni_stirikotnik}
    Skladno z definicijo avtorja~\citeauthor{van2012generalized} je šibki posplošeni 
    štirikotnik definiran kot geometrija $\Gamma = (\PP, \LL, \II)$, za katero velja:
    \begin{enumerate}
        \item Naj bo $\{p, L\}$ antizastava. Potem obstaja natanko ena zastava $\{q, M\}$,
        za katero velja $p \II M \II q \II L$.
        \item Vsaka točka leži na vsaj dveh, a ne vseh, premicah, in dualno, vsaka premica
        nosi vsaj dve, a ne vse, točke.
    \end{enumerate}
    Geometrija $\Gamma$ je posplošeni štirikotnik, 
    če drži prvi aksiom, je "thick" in vsebuje vsaj eno antizastavo.
\end{definicija}

Nas zanimajo zgolj tisti posplošeni štirikotniki, ki so regularni, torej tisti, ki 
so reda $(s, s)$ za nek $s \in \ZZ_+$. Najprej se bomo posvetili klasičnim posplošenim 
štirikotnikom, ki zadoščajo takemu pogoju. Obstajata dve taki družini, in sicer
simplektični posplošeni štirikotniki, ki temeljijo na simplektičnih polarnostih, 
ter ortogonalni posplošeni štirikotniki, ki temeljijo na kvadratičnih formah.

Vendar izkaže se, da sta si, le ta, dualna, torej so si njuni incidenčni grafi 
izomorfni. Iz tega razloga bomo podrobneje predstavili zgolj simplektične posplošene
štirikotnike, o ortogonalnih pa bomo spregovorili zgolj na kratko.

\paragraph{Simplektični posplošeni štirikotniki}
\begin{definicija}\label{def:dualnost_in_polarnost}
    Dualnost $\delta$ projektivnega prostora je permutacija podprostorov v $\PG(n, \mathbb{K})$
    (kjer je $\mathbb{K}$ polje), ki obrne inkluzijo, torej 
    $U \subseteq V \iff \delta(V) \subseteq \delta(U)$,
    za vsaka podprostora $U$ in $V$ v $\PG(n, \mathbb{K})$. 

    Posledično dualnost zamenja podprostore dimenzije $k$ s podprostori dimenzije $n - k - 1$.
    
    Dualnost $\delta$ je polarnost, če je njena inverzna funkcija 
    enaka sami sebi, torej $\delta^2 = \text{id}$. Oziroma lepše povedano, če je 
    involucija (torej reda $2$).

    Če je $\delta$ dualnost, pišemo tudi $\delta(S) = S^\perp$, torej če je $\delta$ polarnost 
    velja $\delta^2 (S) = S^{\perp\perp} = S$ za vsak podprostor $S$.

    Dalje je polarnost simplektična, če je vsak podprostor leži na svojem dualu,
    oziroma $S \subseteq S^\perp$ za vsak podprostor $S$.
\end{definicija}


\begin{zgled}
    Desquarsove projektivne ravnine $\PG(2, \mathbb{K})$ so sami sebi dualne, 
    saj zamenjava točk in premic ohranja inkluzijo.
\end{zgled}

\begin{definicija}\label{def:simplekticni_stirikotnik}
    Naj bo $\mathbb{K}$ polje in $\PG(3, \mathbb{K})$ projektivni prostor 
    dimenzije $3$ nad $\mathbb{K}$.
    
    Naj bo $\tau$ simplektična polarnost, ki preslika točko $(y_0, y_1, y_2, y_3)$
    v ravnino z enačbo $y1 X_0 - y_0 X_1 + y_3 X_2 - y_2 X_3 = 0$.
    Zdaj definiramo simplektični posplošeni štirikotnik $W(\mathbb{K})$ kot geometrijo, 
    kjer so točke $\PP$ točke ($1$-dimenzionalni podprostori) iz $\PG(3, \mathbb{K})$
    in so premice $\LL$ popolnoma izotropne premice iz $\PG(3, \mathbb{K})$ glede na $\tau$ 
    (torej tiste, za katere je $\tau(L) = L$).
\end{definicija}

Lahko se prepričamo, da je simplektični posplošeni štirikotnik, kot definiran zgoraj, 
res posplošeni štirikotnik.

\begin{dokaz}
    Pokazati moramo, da sta aksioma iz definicije posplošenega štirikotnika 
    (\ref{def:posploseni_stirikotnik}) zadoščena. Naj bo, v ta namen, $\{p, L\}$ antizastava
    v $\PG(3, \mathbb{K})$ in $L$ izotropna ($\tau(L) = L$). Poiščimo tako točko $g\II L$
    in izotropno premico $M$, da bosta tako $p$ kot $q$ ležala na njej.

    V $\PG(3, \mathbb{K})$ se premica seka z vsako ravnino, v posebnem se sekata
    tudi premica $L$ in ravnina $p^\tau$. In sekata se natanko v eni točki, saj bi se sicer
    morali sekati v premici $L$ ($L \subset p^\tau$), to pa bi zaradi 
    simplektičnosti pomenilo $p\II L$, kar bi bilo v nasprotju s tem, da je $\{p, L\}$
    antizastava.

    Torej premica $L$ in ravnina $p^\tau$ se sekata v natanko eni točki, ki jo
    označimo z $q$.

    Zdaj pa definiramo premico $M$ kot tisto premico, ki gre skozi $p$ in $q$ (taka je samo ena).
    Pokažemo lahko, da je izotropna, in sicer vemo, da je $q \II p^\tau$, kar pomeni,
    da je $p \II q^\tau$. Ker je polarnost simplektična je tudi $q\II q^\tau$.

    Ker zdaj vemo, da je $p\II q^\tau$ in $q\II q^\tau$ in 
    $(q^\tau \ni p)\II M \II (q\in q^\tau)$ sledi, da tudi cela premica leži v $q^\tau$,
    torej $M^\tau = M$, kar pomeni, da je $M$ izotropna premica.

    Dodatno lahko vidimo še, da skozi vsako točko poteka $|\mathbb{K}| + 1$ premic
    in na vsaki premici leži $|\mathbb{K}| + 1$ točk, torej je red enak 
    $(|\mathbb{K}|, |\mathbb{K}|)$, kar pomeni, da je $W(\mathbb{K})$ posplošeni štirikotnik.
\end{dokaz}

\begin{definicija}\label{def:grassmannove_koordinate}
    Če imamo premico $L$ v $\PG(d, \mathbb{K})$, ki jo določata dve različni točki z 
    homogenimi koordinatami $x = (x_0, x_1, \dots, x_d)$ in $y = (y_0, y_1, \dots, y_d)$,
    so Grassmannove koordinate premice $L$ definirane kot 
    \[p_{ij} = \begin{vmatrix} x_i & x_j \\ y_i & y_j \end{vmatrix},\]
    za vse pare indeksov $0 \leq i < j \leq d$.

    Vseh teh koordinat je $\binom{d+1}{2}$. Zapišemo jih lahko kot eno točko (vektor)
    v večjem projektivnem prostoru $\PG\left(\binom{d+1}{2} - 1, \mathbb{K}\right)$.
\end{definicija}

\begin{opomba}
    Vidimo lahko, da Grassmannove koordinate premice v $W(\mathbb{K})$ 
    zadoščajo enačbi $p_{01} + p_{23} = 0$, in da je vsaka premica iz 
    $\PG(3, \mathbb{K})$, katere Grassmanove koordinate zadoščajo $p_{01} + p_{23} = 0$,
    premica v $W(\mathbb{K})$. To je alternativni način opisa $W(\mathbb{K})$.
\end{opomba}

\paragraph{Ortogonalni posplošeni štirikotniki}

Za razliko od simplektičnih posplošenih štirikotnikov, so ortogonalni posplošeni štirikotniki
$Q(d, \mathbb{K}, q)$ definirani tako, da iz vektorskega prostora $V=\mathbb{K}^{d+1}$ nad poljem $\mathbb{K}$, 
opremljenega s kvadratično formo $q: V \to \mathbb{K}$, Wittovega indeksa $2$, definirano kot 
$q(x_0, \ldots, x_d) = \sum_{0\leq i \leq j \leq d} a_{ij}x_i x_j$, 
izberejo $1$-dimenzionalne podprostore množice $q^{-1}(0)$
kot točke, ter $2$-dimenzionalne podprostore množice $q^{-1}(0)$ kot premice.
Wittov indeks $2$ kvadratične forme $q$ pomeni, da obstajajo $2$-dimenzionalni podprostori v $q^{-1}(0)$,
vendar ne obstajajo višje dimenzionalni podprostori v $q^{-1}(0)$, ustrezen Wittov indeks nam tako zagotovi
ravno pravo bogatost kvadratične forme.

Najenostavnejši primer ortogonalnega enakoredega posplošenega štirikotnika, bi tako dobili z kvadratično 
formo $q(x_0, x_1, x_2, x_3) = x_0x_1 + x_2x_3 + x_4$, recimo nad vektorskem prostoru $ V= \GF(2)^5$. Wittov indeks take forme je res 
enak $2$, saj obstajajo $2$-dimenzionalni podprostori v $q^{-1}(0)$, 
na primer tisti, ki jih določata ravnini $(a, 0, c, 0, 0)$ in $(0, b, 0, d, 0)$. Večdimenzionalnih podprostorov v 
$q^{-1}(0)$ pa ni, saj enačba $q(a, b, c, d, e) = ac + bd = 0$ implicira, da je $e = 0$, 
in da sta vsaj dve izmed ostalih koordinat ničelni.

Taka definicija nam da vedeti, da so posplošeni štirikotniki natanko dovolj bogate algebraične ploskve v 
projektivnih prostorih.

\subsubsection{Posplošeni šestkotniki}
Klasične posplošene šestkotnike imenujmo kot tiste kateri se konstruirajo iz kvadrike
v nekem projektivnem prostoru.

Ob taki definiciji so edini simetrični posplošeni šestkotniki Caylyjevi razdeljeni šestkotniki
$H(s)$ in njihovi duali $H(s)^D$. V nekaterih primerih so si samo dualni, vendar ne v vseh.
Nas bodo taki posplošeni šestkotniki zanimali zgolj do dualnosti natančno, 
saj imajo duali enake, oziroma izomorfne, incidenčne grafe.

\paragraph{Razdeljeni Cayleyevi šestkotniki}

\begin{definicija}
    Definiramo lahko razdeljeni Cayleyev šestkotnik $H(\mathbb{K})$ kot posplošeni šestkotnik, 
    ki ga dobimo z naslednjim načinom.
    Točke $H(\mathbb{K})$ so točke "parabolične" kvadrike $Q(6, \mathbb{K})$ v $\PG(6, \mathbb{K})$ z enačbo
    \[X_0X_4 + X_1X_5 + X_2X_6 = X_3^2.\]
    Premice $H(\mathbb{K})$ pa so dalje tiste premice, ki ležijo na kvadriki $Q(6, \mathbb{K})$
    in katere Grassmannove koordinate zadoščajo naslednjim šestim linearnim enačbam:
    \begin{align*}
        p_{12} &= p_{34}\\
        p_{54} &= p_{32}\\
        p_{20} &= p_{35}\\
        p_{65} &= p_{30}\\
        p_{01} &= p_{36}\\
        p_{46} &= p_{31}
    \end{align*}
    Ta definicija poda ekspliciten opis $H(\mathbb{K})$ na kvadriki $Q(6, \mathbb{K})$.
\end{definicija}

Vendar ime razdeljeni Cayleyev šestkotnik ni naključno, izhaja namreč iz Cayleyeve algebre, ki je algebra
oktonionov, natančneje iz razdeljeni Cayleyeve algebre, ki je algebra oktonionov nad končnimi polji. 
Vredno je predstaviti torej tudi alternativno konstrukcijo prek le te algebre.

Razdeljeni oktonioni $\mathbb{Q}'$ so neasociativna algebra nad $\GF(q)$, z bazo $\{e_0, \ldots, e_7\}$, 
kjer je $e_0$ enota. Seštevanje je komutativno in asociativno, množenje pa je kompleksnejše in podano 
z neko tabelo pravil. Konjugacija elementa $x = \sum_{i=0}^7 x_i e_i$ je definirana 
kot $x^* = x_0 e_0 - \sum_{i=1}^7 x_i e_i$, in imaginarni del elementa $x$
je definiran kot $\text{Im}(x) = \frac{x-x^*}{2} =  \sum_{i=1}^7 x_i e_i$. 
Norma je naravno porodena kot $N(x) = x x^*$.

\begin{definicija}
    Razdeljeni Cayleyev šestkotnik $H(\GF(q))$ lahko definiramo tudi kot posplošeni šestkotnik, 
    kjer so točke $H(\GF(q))$ elementi imaginarnih delov $\text{Im}(\mathbb{Q}')$ iz razdeljenih 
    oktonionov nad $\GF(q)$, za katere velja $N(x) = 0$,
    in premice $H(\GF(q))$ so tisti dvodimenzionalni podprostori, ki ležijo v množici $N^{-1}(0)$.
\end{definicija}

\begin{lema}\label{le:regularnilastne}
Naj bo $G$ $k$-regularen graf in $A$ njegova sosednostna matrika. Potem je $k$ njegova lastna
vrednost in vse ostale lastne vrednosti ležijo na intervalu $[-k, k]$ in so si različne. 
Velja celo to, da je
$k$ enojna lastna vrednost, vendar dokaz tega izpustimo.
\end{lema}
\begin{dokaz}
    Ker je $G$ $k$-regularen, je vsako njegovo vozlišče povezano z $k$ sosedi. Posledično
    se vsaka vrstica sosednostne matrike $A$ sešteje v $k$. 
    Velja torej $A\mathbf{1} = k\mathbf{1}$, $k$ je lastna vrednost.

    Naj bo zdaj $\lambda$ poljubna lastna vrednost matrike $A$, ter $\mathbf{x}$ njen 
    pripadajoč lastni vektor. Naj bo $i^{*}$ indeks, za katerega velja $|x_{i^{*}}| = \max_{j} |x_j|$.
    Potem velja:
    \begin{align*}
        |\lambda| \cdot |x_{i^{*}}| &= |\lambda x_{i^{*}}| = |(A\mathbf{x})_{i^{*}}| = \left|\sum_{j} A_{i^{*}j} x_j\right| \leq \sum_{j} A_{i^{*}j} |x_j|
        \leq \sum_{j} A_{i^{*}j} |x_{i^{*}}| = k |x_{i^{*}}|.
    \end{align*}
    Posledično je $|\lambda| \leq k$.
\end{dokaz}

\begin{definicija}
Graf $G$ je krepko regularen s parametri $(n, k, \lambda, \mu)$, če je
$G$ regularen stopnje $k$ na $n$ vozliščih, ter če imata poljubni dve sosedni

\begin{definicija}
    Naj bosta $(G, \cdot)$ in $(H, \circ)$ grupi. 
    Preslikava $h: G \to H$ je homomorfizem, če za vse $u$ in $v$ v $G$ velja
    $h(u \cdot v) = h(u) \circ h(v)$. Od tod sledi, 
    da $h(e_G) = e_H$, kjer je $e_G$ enota grupe $G$, ter $e_H$ enota grupe $H$ ter
    da $h(u^{-1}) = h(u)^{-1}$ za vsak $u \in G$.
    Če je $h$ bijektivna, potem pravimo, da je $h$ izomorfizem.
    Dve grupi $(G, \cdot)$ in $(H, \circ)$  sta izomorfni, če obstaja izomorfizem
    med njima, pišemo $(G, \cdot) \cong (H, \circ)$.
\end{definicija}
\begin{definicija}
Avtomorfizem grafa $G$ je bijektivna preslikava $\sigma: V(G) \to V(G)$, ki ohranja sosednost, 
torej $uv \in E(G) \implies \sigma(u)\sigma(v) \in E(G)$, oziroma v drugi notaciji 
$u^\sigma v^sigma \in E(G)$. Torej avtomorfizem je izomorfizem grafa $G$ v samega sebe.
Kompozitum dveh avtomorfizmov je zopet avtomorfizem, zato avtomorfizmi grafa $G$,
pod operacijo kompozituma, tvorijo grupo, ki ji pravimo grupa avtomorfizmov 
grafa $G$ in jo označujemo z $\aut(G)$.
V grafu z $n$ vozlišči je $\aut(G) \subset S_n$, kjer je $S_n$ množica vseh permutacij $n$ elementov,
tako velja tudi $|\aut(G)| \leq n!$.
Naj bo dodatno $s \in \aut(G)$ tak, da $s^k = \id$ za nek $k \in \ZZ_+$. 
Potem pravimo, da je $s$ avtomorfizem reda $k$. Posebno, če $k=2$ potem pravimo,
da je $s$ involucija.
\end{definicija}

\begin{definicija}\label{def:vozliscna_tranzitivnost}
Graf $G$ je vozliščno tranzitiven, če je za poljubni vozlišči $u$ in $v$ obstaja avtomorfizem
$\sigma \in \aut(G)$, da $\sigma(u) = v$. Ekvivalentno je to natanko tedaj ko
 grupa avtomorfizmov deluje tranzitivno na množici
vozlišč grafa $G$.
\end{definicija}



\begin{definicija}
Naj bo $G = (V, E)$ graf, kjer je $V$ množica njegovih
vozlišč, ter $E$ množica njegovih povezav. Red grafa $G$ je 
število njegovih vozlišč, torej $\red(G) = |V|$.
\end{definicija}

\begin{definicija}
Naj bo $G = (V, E)$ graf, ter $v \in V$ njegovo vozlišče. 
Stopnja vozlišča $v$ je enaka številu njegovih sosedov, torej
$\deg(v) := \#\{u; \quad u \sim v \land u \in E\}$. Stopnjo 
grafa $G$ pa označimo z maksimalno stopnjo njegovih vozlišč:
$\Delta(G) := \max_{v \in V(G)} \deg(v)$
\end{definicija}

\begin{definicija}
Premer grafa $G$ je največja razdalja med katerima koli 
vozliščema v grafu. Torej, $\diam(G) = \max_{u, v \in V(G)} d(u, v)$, 
kjer je $d : V(G) \times V(G) \rightarrow \N$ metrika, ki meri  
razdaljo med danima vozliščema. 
\end{definicija}

\begin{definicija}
Graf je $\Delta$ regularen, takrat ko je vsako njegovo vozlišče 
stopnje $\Delta$, torej $\forall v \in V(G): \quad \deg(v) = \Delta$.
\end{definicija}

\begin{definicija}
Graf $G = (V, E)$, reda $n$ je krepko regularen natanko tedaj ko je $k$ regularen 
in obstajata 
konstanti $\lambda$ in $\mu$, da imata vsaki sosednji vozlišči 
natanko $\lambda$ skupnih sosedov, vsaki  ne sosedni pa natanko 
$\mu$ skupnih sosedov. Krepko regularen graf z danimi prametri 
označimo s $\srg(n, k, \lambda, \mu)$.
\end{definicija}

\begin{definicija}
Ožina grafa $G$ je enaka dolžini najkrajšega cikla v $G$. Če je graf 
$G$ brez ciklov, označimo ožino grafa z neskončnostjo.
\end{definicija}

\begin{definicija}
Naj bo $G$ povezan acikličen graf na $n$ vozliščih ter $n - 1$ povezavah,

\section{Moorejev graf}\label{sec:moorejev_graf}

Mooreovi grafi so ekstremalni grafi, ki glede na dan premer ter dano 
maksimalno stopnjo dosežejo najbolj optimistično zgornjo mejo za število vozlišč. Leta 1960 sta jih tako 
poimenovala Hoffman in Singelton po Edwardu F. Mooru, ki je prvi zastavil vprašanje o 
opisu in klasifikaciji teh struktur.

\begin{definicija}
Graf $G$, s premerom $D$ in maksimalno stopnjo $\Delta$, 
je Mooreov, če dosega najbolj optimistično zgornjo mejo za število vozlišč -- Mooreovo mejo, podano kot:
\begin{equation}
    1 + \Delta + \Delta (\Delta - 1) + \ldots + \Delta (\Delta - 1)^{D - 1},
\end{equation}
\end{definicija}

\begin{opomba}
Red Moorejevega grafa parametrov $(\Delta, D)$ je večji ali enak redu 
poljubnega grafa $P$, ki zadostuje maksimalni stopnji $\Delta$ ter premeru $D$.
Za poljubno vozlišče $v$ grafa $P$ namreč velja, da je število vozlišč od $v$ 
oddaljenih za $i$ največ toliko kot, če bi bilo vsako vozlišče razdalje $i-1$ 
od $v$ povezano z nadaljnimi $\Delta - 1$ vozlišči, oziroma
$\forall\; 0 \leq i \leq D:\;|N_{i}(v)| \leq |N_{i-1}(v)|\cdot (\Delta - 1)$. Induktivno
tako dobimo zgornjo mejo za red grafa $P$, ki je natanko enaka redu Moorejevega 
grafa.
\end{opomba}

\begin{trditev}\label{tr:regularnost_in_omejitve_dolzine}
Naj bo $G$ Moorejev graf s parametroma $(\Delta, D)$. Tedaj je
$G$ regularen stopnje $\Delta$, ter za poljubni dve vozlišči $u$ in $v$ za 
kateri $u \neq v$ velja, da med njima obstaja natanko ena najkrajša pot 
dolžine največ $D$.
\end{trditev}

\begin{dokaz}
Vzemimo poljubno vozlišče $u$ grafa $G$, $\deg(u)$ stopnja vozlišča $u$,
$\Delta$ pa maksimalna stopnja v grafu $G$. Potem analogno konstrukciji 
Moorejeve spodnje meje velja $|V(G)| \leq 1 + \deg(u)(\Delta - 1) + 
\ldots + \deg(u) (\Delta - 1)^{D - 1}$. Ker pa je Moorejeva meja v grafu
$G$ že dosežena, ter $\deg(u) \leq \Delta$, sledi, da je $\deg(u) = \Delta$.
Podobno se o enoličnosti poti dolžin $D$ prepičamo tako, da opazimo, da 
konstrukcija Moorejeve meje predpostavi, da ima vsako vozlišče $z$ v $i$-tem 
nivoju $N_i(z)$ natanko $\Delta - 1$ sosedov, ki niso povezani z nobenim drugim 
vozliščem iz $i$-tega nivoja, razen ko $i=D$. Ob zanemaritvi notranjih povezav 
zadnjega nivoja $N_D(z)$ skupek nivojev inducira drevesni podgraf za $G$, na katerem 
je vsaka pot med dvema vozliščema dolžine $D$ enolična.
\end{dokaz}

\begin{posledica}\label{po:ozina}
Po trditvi~\ref{tr:regularnost_in_omejitve_dolzine} v grafu s parametroma $(\Delta, D)$ 
med poljubnima vozliščema
$v$ ter $u$ obstaja največ ena najkrajša pot dolžine največ $D$. Posledično 
obstaja tudi pot od $u$ do $v$ dolžine $D + 1$. Najkrajši cikel 
v grafu je tako lahko napet na najmanj $D + (D + 1)$ vozlišč.
Ožina Moorejevega grafa je torej enaka $g=2D + 1$. !!!! utemeljitev ni vredu.

V nadaljevanju bomo zasledili, da so Mooreovi grafi deležni številnih omejitev, 
kar bo utemeljilo tudi njihovo redkost, namreč Mooreovi grafi obstajajo le za:
\begin{align}
\Delta = 1, D = 0; \text{graf $K_1$} \\
&\Delta = 2, D \geq 1; \text{cikli $C_{2D + 1}$} \\
&D = 2, \Delta \in \{1, 3, 7, 57\}; \text{grafi $P_2$, Petersenov graf, Hoffman-Singeltonov graf, neznani graf} \\
\end{align}

\end{posledica}

\subsection{Omejitve lastnih vrednosti}

Primer v katerem je premer $D = 1$ je trivialen, saj iz tega sledi,
da je tudi $\Delta = 1$ ter edini Moorejev graf s takimi parametri enak 
$P_2$. V nadaljnem torej predpostavimo $D, \Delta > 1$.

Tu bi omenil splošen problem in splošno enačbo ... 
\begin{trditev}
Mooreevi grafi zadostujejo naslednji matrični enačbi:
\begin{equation}
\begin{cases} 
G_{\Delta, 0} = I \\
G_{\Delta, 1} = A + I \\
G_{\Delta, i} = A G_{\Delta, i - 1} - (\Delta - 1) G_{\Delta, i - 2} & \text{za } i \geq 2
\end{cases}
\end{equation}
\end{trditev}

\begin{dokaz}
    Bazna primera za $i = 0$ in $i = 1$ sta očitna, saj $G_{\Delta, 0}$ meri število poti
    dolžine 0, torej je enaka identiteti, $G_{\Delta, 1}$ pa meri število poti dolžine
    največ 1, torej je enaka $A + I$. Za indukcijski korak pa opazujmo, da 
    $A G_{\Delta, i - 1}$ meri število poti dolžine največ $i$ med vozliščema $u$ in $v$,
    vključujoč tudi tiste, ki se iz $i-1$ nivoja vrnejo nazaj v $i-2$-gi nivo. 
    Ker pa je graf $G$ regularen stopnje 
    $\Delta$ za vsako vozlišče obstaja natanko $(\Delta - 1) G_{\Delta, i -2}$ 
    takih sprehodov, ki jih ustrezno odštejemo.
\end{dokaz}

\begin{opomba}
    V posebnem primeru za premer $D=2$, Mooreev graf zadođča enačbi:
    \[A^2 + A - (\Delta - 1) I = J,\]
    kjer je $J$ matrika samih enic.
\end{opomba}


\begin{trditev}
Za $A^k = (A^k_{u, v})_{u, v}$ velja:
\begin{equation}
A^k_{u, v} = \# \text{povezav dolžine $k$ med vozliščema $u$ in $v$}
\label{eq:potence_matrike_sosednosti}
\end{equation}
\end{trditev}

\begin{dokaz}
Vzemimo poljubni vozlišči $u$ in $v$ v grafu $G$. Ker velja 
\begin{align}
A^k_{u, v} &= \sum_{z \in V(G) \setminus \{u, v\}} A_{u, z} A^{k - 1}_{z, v} \\
           &= \sum_{z:~ z \sim u \land v \in \reach(u)} A_{u, z} A^{k - 1}_{z, v}
\end{align}
po definiciji matrike sosednosti $A$ ter indukcijski predpostavki res 
štejemo vse povezave dolžine $k$, ki preko vozlišča $z$ vodijo iz $u$ v $v$.
\end{dokaz}

Omejimo se sedaj na problem premera $D=2$ ter stopnje $\Delta > 1$. Predpostavimo,
da dosega Moorejevo mejo, torej je njegov red enak $\Delta^2 + 1$.
(Kar je tudi dimenzija njegove matrike sosednosti).
Ker je $G$ regularen stopnje $\Delta$ in ker med vsakima disjunktnima 
vozliščema obstaja ena sama pot dolžine največ $D$, za njegovo matriko sosednosti 
$A$ velja naslednja enačba: 
\begin{equation}
A^2 + A - (\Delta - 1) I = J
\end{equation}
, kjer je $I$ identiteta in $J$ matrika samih enic.
Matrika $A^2 + A$ namreč meri število poti dolžin največ $D$
\eqref{eq:potence_matrike_sosednosti}. Ker pa je graf 
regularen za vsako vozlišče obstaja natanko $\Delta$ poti dolžine $D$, ki se 
vrnejo nazaj, zato v enačbi odštejemo ustrezen faktor identične matrike.

Preko dane matrične enačbe lahko izračunamo spektralno polje incidenčne 
matrike $A$. Ker je karakteristični polinom za matriko $J$ enak 
$x^{\Delta^2}(x - (\Delta^2 + 1))$, sledijo naslednje enačbe:
\begin{align}
x^2 + x - (\Delta - 1) = \Delta^2 + 1 \label{eq:karakteristicna_1}\\ 
x^2 + 1 - (\Delta - 1) = 0 \label{eq:karakteristicna_2}
\end{align} 
Od koder dobimo lastne vrednosti $\Delta$ (po \eqref{eq:karakteristicna_1})
večkratnosti 1, ter lastni vrednosti $r$ in $s$ enaki 
$(-1 \pm \sqrt{4\Delta - 3})/2$. Več o večkratnosti lastnih vrednosti 
$r$ in $s$ pa nam podata enačbi $a + b = \Delta^2$ ter sled matrike, ki 
je enaka seštevku vseh lastnih vrednosti, po drugi strani pa 
je sled matrike sosednosti očitno enaka 0, torej: 
$\Delta + a r + b s = 0$. 

Če sta $r$ in $s$ iracionalni števili je $a = b$ ter $\Delta = 2$, 
sicer pa sta $r$ in $s$ celi števili, torej $4\Delta - 3 = m^2$ za nek 
$m \in \ZZ_+$. Ob substitucijah $a = \Delta^2 - b$ in 
$\Delta = \frac{m^2 + 3}{4}$ sledi:
\begin{align}
\Delta + a r + b s &= 0 \\
\Leftrightarrow \\
\frac{m^2 + 3}{4} + \left(\left(\frac{m^2 + 3}{4}\right)^2 - b\right) 
\frac{-1 + m}{2} + b \frac{-1 - m}{2} &= 0 \\
\Leftrightarrow \\
\frac{m^5}{32} + \ldots + \left(\frac{3}{4} - \frac{9}{32}\right)  &= 0\\
\Leftrightarrow \\
m^5 + \ldots + 15 &= 0
\end{align}

Preden nadaljujemo si oglejmo pomemben rezultat linearne algebre.

\begin{izrek}[Izrek o celoštevilskih ničlah]
Naj bo $p = a_n x^n + \ldots + a_0$ polinom z celimi koeficienti, kjer 
je vodilni koeficient neničelen. Če ima polinom celoštevilsko ničlo $q$,
potem $q$ deli $a_0$.
\end{izrek}

\begin{dokaz}
Ker je $q$ ničla polinoma $p$ velja $p(q) = a_n q^n + \ldots + 
a_0 = 0$. Po preureditvi dobimo $a_0 = -(a_n q^n + \ldots + a_1 q)$.
Ker je desna stran deljiva z $q$ je z $q$ deljiv tudi $a_0$.
\end{dokaz}

Ker ima enačba $x^2 + 1 - (\Delta - 1) = 0$ cele koeficiente so njene 
racionalne rešitve nujno cele (po izreku).
S tem rezultatom, ter z upoštevanjem, da je $m$ pozitivno celo število,

\subsection{Posplošeni večkotniki}

\begin{definicija}[Večkotnik]
Naj bo $\Gamma = (P, \mathcal{L}, I)$ geometrija ranga $2$. 
Pravimo, da v geometriji $\Gamma$ \emph{obstaja večkotnik dolžine $n$} 
natanko tedaj, ko obstajajo različni elementi 
\[
x_1, x_2, \ldots, x_{2n} \in P \cup \mathcal{L}
\]
za katere velja:
\begin{enumerate}
  \item elementi se izmenjujejo po vrstah, torej $x_{2i-1} \in P$ in $x_{2i} \in \mathcal{L}$ za vsak $i$;
  \item za vsak $i = 1, 2, \ldots, 2n$ velja $x_i \, I \, x_{i+1}$, pri čemer privzamemo $x_{2n+1} = x_1$.
\end{enumerate}
Zaporedju $(x_1, x_2, \ldots, x_{2n})$ pravimo \emph{večkotnik dolžine $n$} 
v geometriji $\Gamma$.
%
Opozarjamo, da obstoj večkotnika v geometriji ni nujno
ekvivalenten obstoju podgeometrije $\Gamma'$, katere incidenčni
graf je izomorfen ciklu $C_n$.
\end{definicija}

\begin{definicija}[Posplošeni večkotnik]
    Naj bo $n$ pozitivno celo število. Šibki posplošeni $n$-kotnik je geometrija
    $\Gamma = (\PP, \LL, I)$ ranga 2, ki zadošča naslednjima pogojema:
    \begin{enumerate}
        \item $\Gamma$ ne vsebuje običajnega $k$~-kotnika (kot podgeometrijo) za $0 \leq k < n$.
        \item Poljubna dva elementa iz $\PP \cup \LL$ sta vsebovana v nekem običajnem $n$~-kotniku
        (kot podgeometrija) v $\Gamma$, \emph{amartmaju}.
    \end{enumerate} 
    Posplošeni $n$~-kotnik je šibki posplošeni $n$~-kotnik $\Gamma$, ki zadošča še dodatnemu pogoju:
    \begin{enumerate}[resume]
        \item V $\Gamma$ obstaja običajen $(n+1)$~-kotnik (kot podgeometrija).
    \end{enumerate}, 
\end{definicija}

\begin{opomba}
    Vsak običajen $n$~-kotnik je tudi šibki posplošen $n$~-kotnik. % 1.3.10 Van Maldeghem.
\end{opomba}
Če posplošene večkotnike raje gledamo z vidika incidenčnih grafov, potem 
lahko zgornjo definicijo preoblikujemo na sledeč način.

    posplošenih $n$~-kotnikov zadošča naslednjima pogojema: 
    \begin{enumerate}
        \item Ožina grafa je natanko $2n$.
        \item Premer grafa je natanko $n$.
    \end{enumerate}
    Graf $\Gamma$ je incidenčni graf posplošenega $n$~-kotnika, če zadošča še dodatnemu pogoju:
    \begin{enumerate}[resume]
        \item Stopnja vsakega vozlišča v $\Gamma$ je vsaj 3.
    \end{enumerate}
\end{definicija}

\begin{definicija}
Naj bo $G$ usmerjen multigraf. Naj bo $\Gamma$ abelova grupa.
Označimo z $D(G)$ množico usmerjenih polrobov (darts) grafa $G$.
Naj bo $\alpha : D(G) \to \Gamma$ funkcija, ki vsakemu polrobu priredi element iz $\Gamma$,
in naj velja
\[
\forall e \in D(G)\colon\quad \alpha(e^{-1}) = \alpha(e)^{-1}.
\]
Preslikavi $\alpha$ pravimo \emph{napetostna preslikava} (voltage assignment).

Dvig grafa $G$ z napetostno preslikavo $\alpha$ je graf $G^\alpha$, definiran kot:
\begin{enumerate}
    \item $V(G^\alpha) = V(G) \times \Gamma$,
    \item $E(G^\alpha) = \{\, \bigl( (u,g), (v,\, g \cdot \alpha(e)) \bigr) \;\mid\; 
           e = uv \in D(G),\ g \in \Gamma \,\}$.
\end{enumerate}
\end{definicija}
